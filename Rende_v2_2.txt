// ── Formatação ────────────────────────────────────────────────────────
function formatCurrency(val) {
  const n = Number(val) || 0;
  return `R$ ${n.toFixed(2).replace(".", ",")}`;
}

function formatQty(val, unit = "") {
  const n = Number(val) || 0;
  const formatted = Number.isInteger(n) ? String(n) : n.toFixed(2).replace(/\.?0+$/, "").replace(".", ",");
  return unit ? `${formatted} ${unit}` : formatted;
}

// ── Regra PEPS: cor/label conforme dias restantes ─────────────────────
function getPepsBadge(validadeDias) {
  if (validadeDias === null || validadeDias === undefined) {
    return { variant: "neutral", label: "Sem validade", urgent: false };
  }
  if (validadeDias <= PEPS_LIMITE_CRITICO) {
    return { variant: "red", label: `Usar hoje (${validadeDias}d)`, urgent: true };
  }
  if (validadeDias <= PEPS_LIMITE_ALERTA) {
    return { variant: "amber", label: `Vencendo logo (${validadeDias}d)`, urgent: true };
  }
  return { variant: "green", label: `${validadeDias}d restantes`, urgent: false };
}

// ── Categoria do mercado → ambiente de armazenamento sugerido ─────────
function mapCategoriaToAmbiente(cat) {
  if (!cat) return "Despensa Seca";
  const c = cat.toLowerCase();
  if (c.includes("horti") || c.includes("latic") || c.includes("frio")) return "Geladeira";
  if (c.includes("açougue") || c.includes("congel")) return "Freezer";
  if (c.includes("limpeza") || c.includes("higiene") || c.includes("bebê") || c.includes("infantil")) {
    return "Limpeza & Bebê";
  }
  return "Despensa Seca";
}

// ── Botijão de gás: datas (AAAA-MM-DD, sempre no horário local) ───────
function parseISODate(s) {
  const [y, m, d] = String(s).split("-").map(Number);
  return new Date(y, (m || 1) - 1, d || 1);
}
function toISODate(date) {
  const mm = String(date.getMonth() + 1).padStart(2, "0");
  const dd = String(date.getDate()).padStart(2, "0");
  return `${date.getFullYear()}-${mm}-${dd}`;
}
function diffDias(a, b) {
  return Math.round((b - a) / 86400000);
}
function addDias(date, n) {
  return new Date(date.getFullYear(), date.getMonth(), date.getDate() + n);
}
function formatDataBR(date, curta = false) {
  const dd = String(date.getDate()).padStart(2, "0");
  const mm = String(date.getMonth() + 1).padStart(2, "0");
  const yy = curta ? String(date.getFullYear()).slice(2) : date.getFullYear();
  return `${dd}/${mm}/${yy}`;
}

// ── Botijão de gás: autonomia média, previsão e situação do termômetro ─
function calcGasStats(compras, agora = new Date()) {
  const ord = (compras || [])
    .filter((c) => c && c.data)
    .sort((a, b) => a.data.localeCompare(b.data));
  const intervalos = ord.slice(1).map((c, i) => diffDias(parseISODate(ord[i].data), parseISODate(c.data)));
  const media = intervalos.length
    ? Math.round(intervalos.reduce((s, x) => s + x, 0) / intervalos.length)
    : null;
  const valorMedio = ord.length ? ord.reduce((s, c) => s + (Number(c.valor) || 0), 0) / ord.length : 0;
  const base = {
    ord,
    intervalos,
    media,
    valorMedio,
    ultima: ord[ord.length - 1] || null,
    minDias: intervalos.length ? Math.min(...intervalos) : null,
    maxDias: intervalos.length ? Math.max(...intervalos) : null,
    custoMensal: media ? (valorMedio / media) * 30 : 0,
    status: "sem-dados",
    diasDesde: null,
    diasRestantes: null,
    pctRestante: null,
    previsao: null,
  };
  if (!base.ultima || !media) return base;

  const hoje = new Date(agora.getFullYear(), agora.getMonth(), agora.getDate());
  const dataUltima = parseISODate(base.ultima.data);
  const diasDesde = Math.max(0, diffDias(dataUltima, hoje));
  const diasRestantes = media - diasDesde;
  let status = "verde";
  if (diasRestantes <= 0) status = "vencido";
  else if (diasRestantes <= GAS_ALERTA_VERMELHO_DIAS) status = "vermelho";
  else if (diasRestantes <= GAS_ALERTA_AMARELO_DIAS) status = "amarelo";
  return {
    ...base,
    status,
    diasDesde,
    diasRestantes,
    pctRestante: Math.min(1, Math.max(0, diasRestantes / media)),
    previsao: addDias(dataUltima, media),
  };
}

// ── Moradores: tipo (pela idade), emoji, idade por extenso e resumo da família
function classificarMembro(m) {
  const bruto = m ? m.idade : null;
  const idade = bruto === null || bruto === undefined || bruto === "" ? NaN : Number(bruto);
  if (Number.isFinite(idade)) {
    const meses = m.idadeUnidade === "meses" ? idade : idade * 12;
    if (meses < 24) return "bebê";
    if (meses < 18 * 12) return "criança";
    if (meses >= 60 * 12) return "idoso";
    return "adulto";
  }
  return (m && m.tipo) || "adulto"; // sem idade: usa o tipo guardado
}
function emojiDoMembro(m) {
  return (m && m.emoji) || TIPOS_MORADOR[classificarMembro(m)].emoji;
}
function formatarIdade(m) {
  const bruto = m ? m.idade : null;
  if (bruto === null || bruto === undefined || bruto === "" || !Number.isFinite(Number(bruto))) return "idade não informada";
  const n = Number(bruto);
  return m.idadeUnidade === "meses" ? `${n} ${n === 1 ? "mês" : "meses"}` : `${n} ${n === 1 ? "ano" : "anos"}`;
}
function descreverComposicao(perfil) {
  const ms = (perfil && perfil.membros) || [];
  const partes = Object.keys(TIPOS_MORADOR)
    .map((tipo) => {
      const grupo = ms.filter((m) => classificarMembro(m) === tipo);
      if (grupo.length === 0) return null;
      const nome = TIPOS_MORADOR[tipo].plural[grupo.length === 1 ? 0 : 1];
      const idades = tipo === "criança" || tipo === "bebê" ? ` (${grupo.map(formatarIdade).join(", ")})` : "";
      return `${grupo.length} ${nome}${idades}`;
    })
    .filter(Boolean);
  return partes.length ? partes.join(", ") : "moradores não informados";
}

// ── Perfis antigos (adultos/criancas/bebes em texto) → lista de moradores
function parseIdadeLegado(txt) {
  const s = String(txt || "").toLowerCase();
  let m = s.match(/(\d+)\s*anos?\s*e\s*(\d+)\s*mes/) || s.match(/(\d+)\s*a\s*(\d+)\s*m\b/);
  if (m) return { idade: Number(m[1]) * 12 + Number(m[2]), idadeUnidade: "meses" };
  m = s.match(/(\d+)\s*mes/);
  if (m) return { idade: Number(m[1]), idadeUnidade: "meses" };
  m = s.match(/(\d+)\s*(anos?|a\b)/);
  if (m) return { idade: Number(m[1]), idadeUnidade: "anos" };
  return { idade: null, idadeUnidade: "anos" };
}
function garantirMembros(p) {
  if (!p || Array.isArray(p.membros)) return p;
  const membros = [];
  const novo = (nome, tipo, idadeInfo) =>
    membros.push({ id: `mem-${membros.length + 1}`, nome, tipo, emoji: null, ...idadeInfo });
  const adultos = Number(p.adultos) || 0;
  for (let i = 0; i < adultos; i++) novo(`Adulto ${i + 1}`, "adulto", { idade: null, idadeUnidade: "anos" });
  [["criancas", "criança", "Criança"], ["bebes", "bebê", "Bebê"]].forEach(([campo, tipo, rotulo]) => {
    const txt = String(p[campo] || "").trim();
    if (!txt || txt === "-") return;
    const qtd = parseInt(txt, 10) || 1;
    for (let i = 0; i < qtd; i++) novo(qtd > 1 ? `${rotulo} ${i + 1}` : rotulo, tipo, parseIdadeLegado(txt));
  });
  const { adultos: _a, criancas: _c, bebes: _b, ...resto } = p;
  return { ...resto, membros };
}

// ── Receitas × despensa: o ingrediente da receita corresponde a este item?
function ingredienteCombinaComItem(ingNome, itemNome) {
  const norm = (s) => String(s || "").toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").trim();
  const a = norm(ingNome);
  const b = norm(itemNome);
  if (!a || !b) return false;
  if (a === b || a.includes(b) || b.includes(a)) return true;
  const pa = a.split(" ")[0];
  return pa.length > 3 && pa === b.split(" ")[0]; // mesma 1ª palavra (ex.: "Tomate")
}

// ── Linguagem: troca termos antigos por expressões do dia a dia (também em dados já salvos)
function suavizarTexto(s) {
  return String(s || "")
    .replace(/ruptura de ciclo/gi, "Costuma faltar")
    .replace(/abaixo do m[ií]nimo/gi, "Acabando");
}

// ── Copiar texto (com plano B quando o navegador bloqueia a área de transferência) ───
async function copiarTexto(texto) {
  try {
    await navigator.clipboard.writeText(texto);
    return true;
  } catch {
    try {
      const ta = document.createElement("textarea");
      ta.value = texto;
      ta.style.position = "fixed";
      ta.style.opacity = "0";
      document.body.appendChild(ta);
      ta.select();
      const ok = document.execCommand("copy");
      document.body.removeChild(ta);
      return ok;
    } catch {
      return false;
    }
  }
}

// ── Validação do banco salvo/importado: completa campos ausentes para o app não quebrar ───
function normalizeDatabase(raw) {
  const base = createInitialDatabase();
  const db = { ...base, ...(raw || {}) };
  ["perfis", "estoque", "marmitas", "receitas", "listaCompras", "compras"].forEach((k) => {
    if (!Array.isArray(db[k])) db[k] = base[k];
  });
  if (db.perfis.length === 0) db.perfis = base.perfis;
  db.centro_custos =
    raw && raw.centro_custos && typeof raw.centro_custos === "object"
      ? { ...base.centro_custos, ...raw.centro_custos }
      : base.centro_custos;
  // Perfis antigos ganham a lista de moradores
  db.perfis = db.perfis.map(garantirMembros);
  // Receitas-exemplo por momento do dia entram uma única vez em bancos antigos
  if (!(raw && raw.exemplosReceitasV21)) {
    const ids = new Set(db.receitas.map((r) => r.id));
    const momentoPorId = Object.fromEntries(base.receitas.map((r) => [r.id, r.momento]));
    db.receitas = [
      ...db.receitas.map((r) => (!r.momento && momentoPorId[r.id] ? { ...r, momento: momentoPorId[r.id] } : r)),
      ...base.receitas.filter((r) => r.id.startsWith("rec-ex-") && !ids.has(r.id)),
    ];
    db.exemplosReceitasV21 = true;
  }
  // Botijão de gás: bancos antigos (sem esse campo) recebem as compras iniciais
  db.gasBotijao =
    raw && raw.gasBotijao && Array.isArray(raw.gasBotijao.compras)
      ? { ...base.gasBotijao, ...raw.gasBotijao }
      : base.gasBotijao;
  if (!db.perfis.some((p) => p.id === db.perfilAtivoId)) db.perfilAtivoId = db.perfis[0].id;
  return db;
}

const buildGeminiUrl = () =>
  `https://generativelanguage.googleapis.com/v1beta/models/${GEMINI_MODEL}:generateContent?key=${GEMINI_API_KEY}`;

// ── Requisição com tentativas automáticas (backoff exponencial) ───────
async function fetchGeminiWithBackoff(url, payload, maxRetries = 3) {
  let delay = 1000;
  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    try {
      const response = await fetch(url, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      });
      if (response.ok) {
        return await response.json();
      }
      if (attempt === maxRetries) throw new Error(`HTTP ${response.status}`);
    } catch (err) {
      if (attempt === maxRetries) throw err;
    }
    await new Promise((r) => setTimeout(r, delay));
    delay *= 2;
  }
}

Classifique cada item em uma das categorias: "Hortifrúti & Sazonais", "Açougue & Congelados", "Laticínios & Frios", "Mercearia & Granel", "Padaria & Lanches", "Infantil & Bebê", "Limpeza & Higiene".
Defina o ambiente: "Despensa Seca", "Geladeira", "Freezer" ou "Limpeza & Bebê".`;

  const parts = [{ text: systemPrompt }];
  if (base64Data && mimeType) {
    parts.push({ inlineData: { mimeType, data: base64Data } });
    parts.push({ text: "Extraia estabelecimento, data, total e todos os itens." });
  } else {
    parts.push({ text: `Extraia estabelecimento, data, total e todos os itens deste texto:\n\n${rawText}` });
  }

  const payload = {
    contents: [{ role: "user", parts }],
    generationConfig: {
      responseMimeType: "application/json",
      responseSchema: {
        type: "OBJECT",
        properties: {
          mercado: { type: "STRING" },
          data: { type: "STRING" },
          total: { type: "NUMBER" },
          itens: {
            type: "ARRAY",
            items: {
              type: "OBJECT",
              properties: {
                nome: { type: "STRING" },
                categoria: { type: "STRING" },
                ambiente: { type: "STRING" },
                quantidade: { type: "NUMBER" },
                unidade: { type: "STRING" },
                precoUnit: { type: "NUMBER" },
                precoTotal: { type: "NUMBER" },
              },
              required: ["nome", "categoria", "ambiente", "quantidade", "unidade", "precoUnit", "precoTotal"],
            },
          },
        },
        required: ["mercado", "data", "total", "itens"],
      },
    },
  };

  const data = await fetchGeminiWithBackoff(apiUrl, payload);
  const text = data?.candidates?.[0]?.content?.parts?.[0]?.text;
  if (!text) throw new Error("Falha ao ler nota");
  return JSON.parse(text);
}

// ── HelpTip: botão "?" que abre um balão curto explicando o cartão ────
function HelpTip({ text, tone = "dark", className = "" }) {
  const [pos, setPos] = useState(null);
  const ref = useRef(null);

  useEffect(() => {
    if (!pos) return undefined;
    const fechar = () => setPos(null);
    const fora = (e) => {
      if (!ref.current || !ref.current.contains(e.target)) fechar();
    };
    const esc = (e) => e.key === "Escape" && fechar();
    document.addEventListener("pointerdown", fora);
    document.addEventListener("keydown", esc);
    window.addEventListener("scroll", fechar, true);
    window.addEventListener("resize", fechar);
    return () => {
      document.removeEventListener("pointerdown", fora);
      document.removeEventListener("keydown", esc);
      window.removeEventListener("scroll", fechar, true);
      window.removeEventListener("resize", fechar);
    };
  }, [pos]);

  const alternar = (e) => {
    e.stopPropagation();
    e.preventDefault();
    if (pos) return setPos(null);
    const r = ref.current.getBoundingClientRect();
    const largura = Math.min(260, window.innerWidth - 24);
    const left = Math.max(12, Math.min(r.right - largura, window.innerWidth - largura - 12));
    setPos({ left, top: r.bottom + 8, width: largura });
  };

  // Usa <span role="button"> (e não <button>) para poder ficar dentro de cartões clicáveis
  return (
    <span ref={ref} className={`inline-flex shrink-0 ${className}`}>
      <span
        role="button"
        tabIndex={0}
        aria-label="O que é isto?"
        aria-expanded={!!pos}
        onClick={alternar}
        onKeyDown={(e) => (e.key === "Enter" || e.key === " ") && alternar(e)}
        className={`w-5 h-5 rounded-full text-[11px] font-bold normal-case tracking-normal flex items-center justify-center cursor-pointer select-none ${
          tone === "light"
            ? "bg-white/20 text-white hover:bg-white/30"
            : "bg-[#EDF5E8] text-[#2D5016] hover:bg-[#DCEBD0]"
        }`}
      >
        ?
      </span>
      {pos && (
        <span
          role="tooltip"
          style={{ position: "fixed", left: pos.left, top: pos.top, width: pos.width, zIndex: 70 }}
          className="block p-3 rounded-xl bg-[#1A1F16] text-white text-xs leading-relaxed font-normal normal-case tracking-normal text-left shadow-xl"
        >
          {text}
        </span>
      )}
    </span>
  );
}

// ── Badge: variant = green | amber | red | neutral | blue ─────────────
function Badge({ children, variant = "green", className = "" }) {
  const styles = {
    green: "bg-[#EDF5E8] text-[#2D5016] border-[#C3DEB0]",
    amber: "bg-[#FEF3DC] text-[#B86B0E] border-[#F2C98A]",
    red: "bg-[#FDECEA] text-[#C0392B] border-[#F5B7B1]",
    neutral: "bg-[#F2EDE4] text-[#5C6652] border-[#DDD8CC]",
    blue: "bg-[#EBF3FA] text-[#1F5C8B] border-[#B8D4EC]",
  };
  return (
    <span
      className={`inline-flex items-center gap-1 px-2.5 py-0.5 rounded-full text-[11px] font-semibold border ${
        styles[variant] || styles.neutral
      } ${className}`}
    >
      {children}
    </span>
  );
}

// Dados em db.gasBotijao.compras = [{ id, data: "AAAA-MM-DD", valor }]
function GasBotijaoCard({ gas, setGas, notify }) {
  // ── ESTADO LOCAL ────────────────────────────────────────────────────
  const [modalAberto, setModalAberto] = useState(false);
  const [verHistorico, setVerHistorico] = useState(false);
  const [formData, setFormData] = useState(toISODate(new Date()));
  const [formValor, setFormValor] = useState("");

  // ── DADOS DERIVADOS ─────────────────────────────────────────────────
  const stats = calcGasStats(gas.compras);
  const info = GAS_STATUS[stats.status];
  const pct = stats.pctRestante ?? 0;
  const diasTotal = stats.media || 1;
  // Faixas do termômetro (da esquerda = comprou hoje, até a direita = acabou)
  const pctVermelho = Math.min(100, (GAS_ALERTA_VERMELHO_DIAS / diasTotal) * 100);
  const pctAmarelo = Math.min(100 - pctVermelho, ((GAS_ALERTA_AMARELO_DIAS - GAS_ALERTA_VERMELHO_DIAS) / diasTotal) * 100);
  const pctVerde = Math.max(0, 100 - pctAmarelo - pctVermelho);
  const posMarcador = stats.diasDesde == null ? 0 : Math.min(100, (stats.diasDesde / diasTotal) * 100);
  const historico = [...stats.ord].reverse();

  let titulo = "Aguardando a 2ª troca";
  let subtitulo = "Registre mais uma troca e o Rende! calcula quanto o botijão dura em casa.";
  if (stats.status !== "sem-dados") {
    if (stats.diasRestantes > 0) {
      titulo = `~${stats.diasRestantes} ${stats.diasRestantes === 1 ? "dia" : "dias"} restantes`;
      subtitulo = `${stats.diasDesde} ${stats.diasDesde === 1 ? "dia" : "dias"} desde a última troca (${formatDataBR(parseISODate(stats.ultima.data))})`;
    } else if (stats.diasRestantes === 0) {
      titulo = "Troca prevista para hoje";
      subtitulo = `Última troca em ${formatDataBR(parseISODate(stats.ultima.data))}`;
    } else {
      const atraso = Math.abs(stats.diasRestantes);
      titulo = `${atraso} ${atraso === 1 ? "dia" : "dias"} além da média`;
      subtitulo = `Última troca em ${formatDataBR(parseISODate(stats.ultima.data))} — vale ter um botijão reserva`;
    }
  }

  // ── AÇÕES ───────────────────────────────────────────────────────────
  const abrirModal = () => {
    setFormData(toISODate(new Date()));
    setFormValor(stats.ultima ? String(stats.ultima.valor) : "");
    setModalAberto(true);
  };

  const salvarCompra = (e) => {
    e.preventDefault();
    const valor = parseFloat(String(formValor).replace(",", "."));
    if (!formData || !(valor > 0)) {
      notify("Informe a data e o valor pago pelo botijão.");
      return;
    }
    if (gas.compras.some((c) => c.data === formData)) {
      notify("Já existe uma compra registrada nessa data.");
      return;
    }
    setGas((prev) => ({
      ...prev,
      compras: [...prev.compras, { id: `gas-${formData}-${Date.now()}`, data: formData, valor }],
    }));
    setModalAberto(false);
    notify("Troca de botijão registrada! O termômetro foi reiniciado.");
  };

  const removerCompra = (id) => {
    setGas((prev) => ({ ...prev, compras: prev.compras.filter((c) => c.id !== id) }));
    notify("Compra de botijão removida.");
  };

  // ── RENDER ──────────────────────────────────────────────────────────
  return (
    <div className="bg-white rounded-2xl p-4 sm:p-5 md:p-6 border border-[#DDD8CC] shadow-xs space-y-4">
      {/* Cabeçalho: título, situação e botão de registro */}
      <div className="flex flex-wrap items-start justify-between gap-2">
        <div className="min-w-0">
          <h3 className="font-serif-editorial text-lg font-bold text-[#2D5016] flex items-center gap-2">
            <Flame className="w-5 h-5 text-[#D4821A]" />
            Botijão de Gás
              <HelpTip text={AJUDA.botijao} />
          </h3>
          <p className="text-xs text-[#6B7A5E]">Autonomia média e previsão da próxima troca.</p>
        </div>
        <div className="flex flex-wrap items-center gap-2">
          <Badge variant={info.badge}>{info.label}</Badge>
          <button
            onClick={abrirModal}
            className="px-3 py-1.5 rounded-xl bg-[#D4821A] hover:bg-[#B86B0E] text-white text-xs font-semibold flex items-center gap-1.5"
          >
            <Plus className="w-3.5 h-3.5" /> Registrar troca
          </button>
        </div>
      </div>

      {/* Botijão (nível restante) + resumo em texto */}
      <div className="flex items-center gap-4">
        <svg viewBox="0 0 48 76" className="w-12 h-[72px] shrink-0" aria-hidden="true">
          <defs>
            <clipPath id="gas-body-clip">
              <rect x="6" y="18" width="36" height="54" rx="12" />
            </clipPath>
          </defs>
          <path d="M14 18 V10 a4 4 0 0 1 4 -4 h12 a4 4 0 0 1 4 4 V18" fill="none" stroke="#5C6652" strokeWidth="3" strokeLinecap="round" />
          <rect x="20" y="1.5" width="8" height="5" rx="1.5" fill="#5C6652" />

          </g>
          <path d="M13 30 V60" stroke="#fff" strokeWidth="3" strokeLinecap="round" opacity="0.55" />
        </svg>
        <div className="min-w-0">
          <div className="font-serif-editorial text-xl sm:text-2xl font-extrabold" style={{ color: info.cor }}>
            {titulo}
          </div>
          <p className="text-xs text-[#6B7A5E] mt-0.5">{subtitulo}</p>
        </div>
      </div>

      {/* Termômetro: verde → âmbar → vermelho; a bolinha mostra onde estamos hoje */}

        <div>

            </div>
            <div
              className="absolute top-1/2 w-4 h-4 rounded-full bg-white border-[3px] shadow-sm"
              style={{ left: `${posMarcador}%`, borderColor: info.cor, transform: "translate(-50%, -50%)" }}
            />
          </div>
          <div className="flex justify-between text-[10px] text-[#6B7A5E] mt-1.5">
            <span>Dia 0 (troca)</span>
            <span>Dia {stats.media} (média)</span>

        </div>
      )}

      {/* Números-chave */}
      <div className="grid grid-cols-3 gap-2">
        <div className="p-2.5 rounded-xl bg-[#FDF8F0] border border-[#EAE4D9]">

        <div className="p-2.5 rounded-xl bg-[#FDF8F0] border border-[#EAE4D9]">

        >

        {verHistorico && (

            {historico.map((c, i) => {

                step="0.01"

                placeholder="Ex.: 114,00"
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-sm focus:outline-hidden focus:border-[#2D5016]"
                required
              />
            </div>
          </div>
          <button
            type="submit"
            className="w-full px-4 py-2.5 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] text-white text-xs font-bold"

function InicioTab({

  notify,

  loadingAudit,

}) {

  const [missingQtd, setMissingQtd] = useState(1);
  const [missingUn, setMissingUn] = useState("un");

  // ── DADOS DERIVADOS: KPIs da despensa ───────────────────────────────
  const itensDisponiveis = useMemo(
    () => estoque.filter((i) => Number(i.quantidade) > 0).length,
    [estoque]
  );

  const itensRepor = useMemo(
    () => estoque.filter((i) => Number(i.quantidade) <= Number(i.estoqueMinimo)),
    [estoque]
  );

  const totalMarmitas = useMemo(
    () => marmitas.reduce((acc, m) => acc + (Number(m.porcoes) || 0), 0),
    [marmitas]
  );

  const autonomiaDias = useMemo(() => {
    const baseEstoque = Math.max(2, Math.round((itensDisponiveis / Math.max(1, estoque.length)) * 12));
    const bonusFreezer = Math.round(totalMarmitas * 0.45);
    return baseEstoque + bonusFreezer;
  }, [itensDisponiveis, estoque.length, totalMarmitas]);

  // ── DADOS DERIVADOS: Financeiro & centro de custos ──────────────────
  const valorImobilizadoDespensa = useMemo(
    () =>
      estoque.reduce(
        (acc, i) => acc + (Number(i.quantidade) || 0) * (Number(i.precoMedio) || 0),
        0
      ),
    [estoque]
  );

  const itensRuptura = useMemo(
    () => estoque.filter((i) => i.rupturaCiclo),
    [estoque]
  );

  const orcamentoPct = useMemo(() => {
    const teto = Number(perfilAtivo?.orcamentoAlimentacao) || 1800;
    return Math.min(100, Math.round((centroCustos.gastoAcumuladoMes / teto) * 100));
  }, [perfilAtivo, centroCustos]);

  // ── DADOS DERIVADOS: Radar PEPS (5 menores validades com estoque > 0) ───
  const radarPeps = useMemo(() => {
    return [...estoque]
      .filter(
        (i) => Number(i.quantidade) > 0 && i.validadeDias !== null && Number(i.validadeDias) <= PEPS_LIMITE_RADAR
      )
      .sort((a, b) => (a.validadeDias ?? 999) - (b.validadeDias ?? 999))
      .slice(0, 5);
  }, [estoque]);

  // ── DADOS DERIVADOS: receitas que aproveitam o que vence primeiro ────
  // Escolhe até 3 receitas que, juntas, cobrem o máximo de itens perto de vencer.
  const { sugestoesRadar, semReceitaRadar } = useMemo(() => {
    const peso = (it) => 1 + (PEPS_LIMITE_RADAR - Math.min(Number(it.validadeDias), PEPS_LIMITE_RADAR)) / PEPS_LIMITE_RADAR;
    const candidatas = (receitas || [])
      .map((rec) => ({
        rec,
        usa: radarPeps.filter((it) => (rec.ingredientes || []).some((ing) => ingredienteCombinaComItem(ing.nome, it.nome))),
      }))
      .filter((c) => c.usa.length > 0);
    let restantes = [...radarPeps];
    const escolhidas = [];
    while (escolhidas.length < 3) {
      const melhor = candidatas
        .filter((c) => !escolhidas.includes(c))
        .map((c) => ({ ...c, novos: c.usa.filter((i) => restantes.includes(i)) }))
        .filter((c) => c.novos.length > 0)
        .sort((a, b) => b.novos.reduce((s, i) => s + peso(i), 0) - a.novos.reduce((s, i) => s + peso(i), 0))[0];
      if (!melhor) break;
      escolhidas.push(melhor);
      restantes = restantes.filter((i) => !melhor.novos.includes(i));
    }
    return { sugestoesRadar: escolhidas, semReceitaRadar: restantes };
  }, [radarPeps, receitas]);

    e.preventDefault();
    if (!missingNome.trim()) return;
    onRegisterMissingItem({
      nome: missingNome.trim(),
      categoria: missingCat,
      quantidade: Math.max(1, Number(missingQtd) || 1),
      unidade: missingUn,
    });
    setMissingNome("");
    setMissingQtd(1);
    setMissingModal(false);
  };

  // ── RENDER ──────────────────────────────────────────────────────────
  return (
    <div className="space-y-6">
      {/* HERO OPERACIONAL DA DESPENSA */}
      <div
        className="rounded-2xl p-4 sm:p-6 md:p-7 text-white shadow-lg"
        style={{
          background: `linear-gradient(135deg, ${COLORS.green} 0%, ${COLORS.greenMid} 100%)`,
        }}
      >
        <div className="flex flex-col lg:flex-row lg:items-center justify-between gap-5">
          <div className="min-w-0">
            {/* Nome da família + moradores em emojis */}
            <h2 className="font-serif-editorial text-2xl md:text-3xl font-extrabold tracking-tight">
              {perfilAtivo.nome}
            </h2>
            <div
              className="text-2xl md:text-3xl mt-1 leading-tight tracking-wide"

            >

                ? perfilAtivo.membros.map(emojiDoMembro).join(" ")

              <span className="font-semibold text-white">Foco do mês:</span>{" "}

            <p className="font-serif-editorial text-lg md:text-xl font-bold text-white mt-5">
              Como está a casa hoje
            </p>
          </div>

          <button
            onClick={() => setMissingModal(true)}
            className="px-4 py-3 rounded-xl bg-[#D4821A] hover:bg-[#B86B0E] text-white text-xs md:text-sm font-bold shadow-md flex items-center gap-2 self-start lg:self-center transition-all shrink-0"
          >
            <Plus className="w-4 h-4" />
            Acabou algo?
          </button>
        </div>

        {/* 4 KPIs PRINCIPAIS FOCADOS NA DESPENSA */}
        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3.5 mt-6">
          <button
            onClick={() => onGoToTab("despensa", "todos")}
            className="relative text-left p-3 sm:p-4 rounded-xl bg-white/10 hover:bg-white/15 border border-white/15 transition-all"
          >
            <HelpTip tone="light" className="absolute top-2 right-2" text={AJUDA.temEmCasa} />
            <span className="text-[11px] uppercase tracking-wider text-white/75 block pr-6">
              Tem em casa
            </span>

              {itensDisponiveis}
              <span className="text-sm font-normal text-white/70">/{estoque.length}</span>

            <span className="text-[11px] text-[#A8E06B] mt-1 block">
              Produtos com saldo na despensa
            </span>
          </button>

          <button
            onClick={() => onGoToTab("despensa", "repor")}
            className="relative text-left p-3 sm:p-4 rounded-xl bg-white/10 hover:bg-white/15 border border-white/15 transition-all"
          >
            <HelpTip tone="light" className="absolute top-2 right-2" text={AJUDA.pedindoReposicao} />
            <span className="text-[11px] uppercase tracking-wider text-white/75 block pr-6">

        {/* RADAR PEPS */}

              <div className="h-2 bg-[#EAE4D9] rounded-full overflow-hidden mt-2">

            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
              <div>
                <h4 className="text-xs font-bold text-[#C0392B] flex items-center gap-1.5">
                  <AlertTriangle className="w-4 h-4" />
                  {itensRuptura.length} itens acabaram antes da hora
                </h4>
                <p className="text-[11px] text-[#1A1F16] mt-0.5">
                  Esses produtos faltaram no meio do mês e precisaram de compra extra (
                  <strong>{formatCurrency(centroCustos.gastoRecomprasEmergenciais)}</strong>), geralmente mais cara que na compra principal.
                </p>
              </div>
              <button
                onClick={onAddAllRupturaToMarket}
                className="px-3 py-1.5 rounded-lg bg-[#C0392B] hover:bg-[#A93226] text-white text-[11px] font-semibold shrink-0"
              >
                Reforçar na próxima lista
              </button>
            </div>

            <div className="flex flex-wrap gap-1.5 pt-1">
              {itensRuptura.map((r) => (
                <span
                  key={r.id}
                  className="px-2.5 py-1 rounded-lg bg-white border border-[#F5B7B1] text-[11px] font-medium text-[#1A1F16]"
                >
                  {r.nome} ({formatQty(r.quantidade, r.unidade)} / mín {r.estoqueMinimo})

              ))}
            </div>
          </div>
        </div>
      </div>

      {/* BOTIJÃO DE GÁS: autonomia média + termômetro de troca */}
      <GasBotijaoCard gas={gasBotijao} setGas={setGasBotijao} notify={notify} />

      {/* AUDITORIA ESTRATÉGICA COM GEMINI API */}
      <div className="bg-white rounded-2xl p-4 sm:p-5 md:p-6 border border-[#DDD8CC] shadow-xs">
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 mb-3">
          <div>
            <h3 className="font-serif-editorial text-lg font-bold text-[#2D5016] flex items-center gap-2">
              <Sparkles className="w-5 h-5 text-[#D4821A]" />
              Dicas Inteligentes para a Casa
              <HelpTip text={AJUDA.dicasCasa} />
            </h3>
            <p className="text-xs text-[#6B7A5E]">
              Cruza sua despensa, o freezer e a idade das crianças para sugerir melhorias.
            </p>
          </div>
          <button
            onClick={onRunAudit}
            disabled={loadingAudit}
            className="px-4 py-2 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] disabled:opacity-50 text-white text-xs font-semibold flex items-center gap-2 self-start"
          >
            <RefreshCw className={`w-3.5 h-3.5 ${loadingAudit ? "animate-spin" : ""}`} />
            {loadingAudit ? "Analisando sua casa..." : analiseIA ? "Atualizar dicas" : "Analisar minha casa"}
          </button>
        </div>

        {analiseIA ? (
          <div className="grid grid-cols-1 md:grid-cols-3 gap-4 pt-2">
            <div className="p-4 rounded-xl bg-[#EDF5E8] border border-[#C3DEB0] space-y-2">
              <div className="flex items-center justify-between">
                <span className="text-xs font-bold text-[#2D5016]">Visão Geral</span>
                {analiseIA.economiaEstimadaMensal > 0 && (
                  <Badge variant="green">
                    -{formatCurrency(analiseIA.economiaEstimadaMensal)}/mês
                  </Badge>
                )}
              </div>
              <p className="text-xs text-[#1A1F16] leading-relaxed">
                {analiseIA.diagnosticoGeral}

            </div>

            <div className="p-4 rounded-xl bg-[#FDF8F0] border border-[#DDD8CC] space-y-1.5">
              <h4 className="text-xs font-bold text-[#D4821A]">
                📦 Para não faltar no meio do mês
              </h4>
              <ul className="space-y-1 text-xs text-[#1A1F16]">
                {analiseIA.ajustesEstoqueMinimo?.map((a, idx) => (

                ))}
              </ul>
            </div>

            <div className="p-4 rounded-xl bg-[#FDF8F0] border border-[#DDD8CC] space-y-1.5">
              <h4 className="text-xs font-bold text-[#2D5016]">
                👶 No prato das crianças
              </h4>
              <ul className="space-y-1 text-xs text-[#1A1F16]">
                {analiseIA.dicasNutricaoCriancas?.map((d, idx) => (
                  <li key={idx}>• {d}</li>
                ))}
              </ul>
            </div>
          </div>
        ) : (
          <div className="p-4 rounded-xl bg-[#FDF8F0] border border-dashed border-[#DDD8CC] text-xs text-[#6B7A5E] text-center">
            Toque em <strong>"Analisar minha casa"</strong> para receber ideias de quantidade para não faltar nada no meio do mês, aproveitamento de receitas e sugestões para as crianças.
          </div>
        )}
      </div>

      {/* MODAL ATALHO RÁPIDO: REGISTRAR ITEM FALTANTE */}
      <Modal
        open={missingModal}
        onClose={() => setMissingModal(false)}
        title="Acabou algo?"

      >
        <form onSubmit={handleCreateMissing} className="space-y-4">
          <div>
            <label className="block text-xs font-semibold text-[#1A1F16] mb-1">
              Qual produto acabou ou está faltando? *
            </label>
            <input
              autoFocus
              required
              type="text"
              value={missingNome}
              onChange={(e) => setMissingNome(e.target.value)}

          <div className="grid grid-cols-3 gap-2.5">
            <div className="col-span-3 sm:col-span-1">

                Quantidade a repor
              </label>
              <input
                type="number"

                value={missingQtd}
                onChange={(e) => setMissingQtd(e.target.value)}
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-sm"
              />
            </div>
            <div className="col-span-3 sm:col-span-2">
              <label className="block text-xs font-semibold text-[#1A1F16] mb-1">

              </label>
              <select
                value={missingCat}

          <div className="flex justify-end gap-2 pt-2">
            <button

              className="px-4 py-2 rounded-xl text-xs font-semibold text-[#6B7A5E]"

              Zerar e enviar para a lista

  );

//    Cards de estoque, filtros, inventário prévio, cadastro/edição e comando por voz.

function DespensaVozTab({
  estoque,
  setEstoque,
  initialFilter = "todos",
  onAddToMarket,
  notify,
}) {
  // ── ESTADO LOCAL: filtros e inventário ──────────────────────────────
  const [ambienteFiltro, setAmbienteFiltro] = useState("Todos");
  const [statusFiltro, setStatusFiltro] = useState(initialFilter);
  const [busca, setBusca] = useState("");
  const [inventarioPrevio, setInventarioPrevio] = useState(false);
  const [checkedPrevio, setCheckedPrevio] = useState({});

  // ── ESTADO LOCAL: voz ───────────────────────────────────────────────
  const [isListening, setIsListening] = useState(false);
  const [voiceTranscript, setVoiceTranscript] = useState("");
  const [loadingVoiceAI, setLoadingVoiceAI] = useState(false);
  const recognitionRef = useRef(null);

  // ── ESTADO LOCAL: modal de cadastro/edição ──────────────────────────
  const [itemModalOpen, setItemModalOpen] = useState(false);
  const [editingItem, setEditingItem] = useState(null);
  const [formItem, setFormItem] = useState({
    nome: "",
    categoria: "Mercearia & Granel",
    ambiente: "Despensa Seca",
    quantidade: 1,
    unidade: "un",
    estoqueMinimo: 1,
    validadeDias: 30,
    precoMedio: 0,
    rupturaCiclo: false,
    sazonalGranel: false,
  });

  // ── EFEITOS ─────────────────────────────────────────────────────────
  useEffect(() => {
    if (initialFilter) setStatusFiltro(initialFilter);
  }, [initialFilter]);

  // ── AÇÕES: microfone (Web Speech API) ───────────────────────────────
  const toggleVoiceRecording = () => {
    const SpeechRecognition =
      window.SpeechRecognition || window.webkitSpeechRecognition;

    if (!SpeechRecognition) {
      notify("Seu navegador não suporta microfone nativo. Digite o comando abaixo e clique em Processar!");
      return;
    }

    if (isListening) {
      recognitionRef.current?.stop();
      setIsListening(false);
      return;
    }

    const recognition = new SpeechRecognition();
    recognition.lang = "pt-BR";
    recognition.continuous = false;
    recognition.interimResults = true;

    recognition.onstart = () => {
      setIsListening(true);
      notify("Microfone ativo! Fale ex: 'Comprei 6 leites e acabei com o óleo'");
    };

    recognition.onresult = (event) => {
      let text = "";
      for (let i = 0; i < event.results.length; i++) {
        text += event.results[i][0].transcript;
      }
      setVoiceTranscript(text);
    };

    recognition.onerror = () => {
      setIsListening(false);
      notify("Não foi possível captar áudio. Você também pode digitar o comando.");
    };

    recognition.onend = () => {
      setIsListening(false);
    };

    recognitionRef.current = recognition;
    recognition.start();
  };

  // ── AÇÕES: comando de voz → IA Gemini → movimentos de estoque ───────
  const applyVoiceCommand = async () => {
    const text = voiceTranscript.trim();
    if (!text) return;
    setLoadingVoiceAI(true);

    try {
      const movimentos = await parseVoiceInventoryWithGemini(text, estoque);
      if (movimentos.length > 0) {
        setEstoque((prev) => {
          const next = [...prev];
          movimentos.forEach((mov) => {
            const idx = next.findIndex(
              (item) =>
                item.nome.toLowerCase().includes(mov.nomeItem.toLowerCase()) ||
                mov.nomeItem.toLowerCase().includes(item.nome.toLowerCase().split(" ")[0])
            );
            if (idx >= 0) {
              const currentQtd = Number(next[idx].quantidade) || 0;
              let newQtd = currentQtd;
              if (mov.operacao === "entrada") {
                newQtd = Number((currentQtd + Number(mov.quantidade)).toFixed(2));
              } else if (mov.operacao === "saida") {
                newQtd =
                  mov.quantidade >= 999
                    ? 0
                    : Math.max(0, Number((currentQtd - Number(mov.quantidade)).toFixed(2)));
              } else {
                newQtd = Math.max(0, Number(mov.quantidade));
              }
              next[idx] = { ...next[idx], quantidade: newQtd };
            } else if (mov.operacao === "entrada" || mov.operacao === "definir") {
              next.unshift({
                id: `est-voz-${Date.now()}-${Math.random().toString(36).slice(2, 5)}`,
                nome: mov.nomeItem,
                categoria: "Mercearia & Granel",
                ambiente: AMBIENTES.includes(mov.ambiente) ? mov.ambiente : "Despensa Seca",
                quantidade: Number(mov.quantidade) || 1,
                unidade: mov.unidade || "un",
                estoqueMinimo: 1,
                validadeDias: 30,
                precoMedio: 0,
                rupturaCiclo: false,
                sazonalGranel: false,
              });
            }
          });
          return next;
        });
        notify(`${movimentos.length} atualização(ões) aplicada(s) por voz!`);
        setVoiceTranscript("");
      } else {
        notify("Nenhum item identificado no comando. Tente mencionar o nome e a quantidade.");
      }
    } catch {
      // Plano B offline: interpreta frases simples sem IA
      // (CORREÇÃO: o contador agora é calculado ANTES de atualizar o estado)
      const lower = text.toLowerCase();
      const idsAfetados = new Set(
        estoque
          .filter((item) => {
            const firstWord = item.nome.toLowerCase().split(" ")[0];
            return firstWord.length > 3 && lower.includes(firstWord);
          })
          .map((item) => item.id)
      );
      const updatedCount = idsAfetados.size;
      if (updatedCount > 0) {
        setEstoque((prev) =>
          prev.map((item) => {
            if (!idsAfetados.has(item.id)) return item;
            if (lower.includes("acab") || lower.includes("zer") || lower.includes("usei")) {
              return { ...item, quantidade: 0 };
            }
            const matchNum = lower.match(/(\d+([.,]\d+)?)/);
            const delta = matchNum ? parseFloat(matchNum[1].replace(",", ".")) : 1;
            return { ...item, quantidade: Number((Number(item.quantidade) + delta).toFixed(2)) };
          })
        );
        notify(`Atualização rápida aplicada em ${updatedCount} item(ns)!`);
        setVoiceTranscript("");
      } else {
        notify("Não foi possível conectar à IA agora. Use os botões + e - nos cards.");
      }
    }

    setLoadingVoiceAI(false);
  };

  // ── AÇÕES: ajuste de quantidade e modal de cadastro/edição ──────────
  const adjustQty = (id, direction) => {
    setEstoque((prev) =>
      prev.map((item) => {
        if (item.id !== id) return item;
        const step =
          item.unidade === "kg" ||
          item.unidade === "L" ||
          item.unidade === "pct" ||
          item.unidade === "lata" ||
          item.unidade === "pote" ||
          item.unidade === "bdj"
            ? 0.5
            : 1;
        const next = Math.max(0, Number((Number(item.quantidade) + direction * step).toFixed(2)));
        return { ...item, quantidade: next };
      })
    );
  };

  const openModal = (item = null) => {
    if (item) {
      setEditingItem(item);
      setFormItem({ ...item });
    } else {
      setEditingItem(null);
      setFormItem({
        nome: "",
        categoria: "Mercearia & Granel",
        ambiente: "Despensa Seca",
        quantidade: 1,
        unidade: "un",
        estoqueMinimo: 1,
        validadeDias: 30,
        precoMedio: 0,
        rupturaCiclo: false,
        sazonalGranel: false,
      });
    }
    setItemModalOpen(true);
  };

  const saveFormItem = (e) => {
    e.preventDefault();
    if (!formItem.nome.trim()) return;
    const cleaned = {
      ...formItem,
      nome: formItem.nome.trim(),
      quantidade: Math.max(0, Number(formItem.quantidade) || 0),
      estoqueMinimo: Math.max(0, Number(formItem.estoqueMinimo) || 0),
      validadeDias: Number(formItem.validadeDias) || 30,
      precoMedio: Math.max(0, Number(formItem.precoMedio) || 0),
    };

    if (editingItem) {
      setEstoque((prev) =>
        prev.map((i) => (i.id === editingItem.id ? { ...i, ...cleaned } : i))
      );
      notify(`"${cleaned.nome}" atualizado!`);
    } else {
      setEstoque((prev) => [{ id: `est-${Date.now()}`, ...cleaned }, ...prev]);
      notify(`"${cleaned.nome}" cadastrado na despensa!`);
    }
    setItemModalOpen(false);
  };

  // ── DADOS DERIVADOS: lista filtrada ─────────────────────────────────
  const filteredEstoque = useMemo(() => {
    return estoque.filter((item) => {
      if (ambienteFiltro !== "Todos" && item.ambiente !== ambienteFiltro) return false;
      if (busca.trim() && !item.nome.toLowerCase().includes(busca.toLowerCase())) return false;

      const isRepor = Number(item.quantidade) <= Number(item.estoqueMinimo);
      const peps = getPepsBadge(item.validadeDias);

      if (statusFiltro === "repor" && !isRepor) return false;
      if (statusFiltro === "peps" && !peps.urgent) return false;
      if (statusFiltro === "ruptura" && !item.rupturaCiclo) return false;
      if (statusFiltro === "ok" && (isRepor || peps.urgent)) return false;

      return true;
    });
  }, [estoque, ambienteFiltro, statusFiltro, busca]);

  // ── RENDER ──────────────────────────────────────────────────────────
  return (
    <div className="space-y-5">
      {/* PAINEL DE COMANDO POR VOZ (WEB SPEECH API + GEMINI) */}
      <div className="bg-white rounded-2xl p-4 sm:p-5 border border-[#DDD8CC] shadow-xs space-y-3">
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
          <div>
            <h2 className="font-serif-editorial text-xl font-bold text-[#2D5016] flex items-center gap-2">
              <Mic className="w-5 h-5 text-[#D4821A]" />
              Sua Despensa
            </h2>
            <p className="text-xs text-[#6B7A5E]">
              Atualize o que entrou ou saiu tocando nos botões ou falando pelo microfone.
            </p>
          </div>

          <div className="flex flex-wrap items-center gap-2">
            <button
              onClick={() => setInventarioPrevio(!inventarioPrevio)}
              className={`px-3.5 py-2 rounded-xl text-xs font-semibold border flex items-center gap-1.5 transition-all ${
                inventarioPrevio
                  ? "bg-[#D4821A] text-white border-[#D4821A]"
                  : "bg-[#FDF8F0] text-[#1A1F16] border-[#DDD8CC]"
              }`}
            >
              <ClipboardList className="w-4 h-4" />
              {inventarioPrevio ? "Fechar conferência" : "Conferir antes de sair"}
            </button>

            <button
              onClick={() => openModal()}
              className="px-4 py-2 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] text-white text-xs font-semibold flex items-center gap-1.5"
            >
              <Plus className="w-4 h-4" /> Novo item
            </button>
          </div>
        </div>

        {/* BARRA DE VOZ */}
        <div className="p-3.5 rounded-xl bg-[#FDF8F0] border border-[#DDD8CC] flex flex-col sm:flex-row gap-2.5">
          <button
            type="button"
            onClick={toggleVoiceRecording}
            className={`px-4 py-2.5 rounded-xl text-xs font-bold flex items-center justify-center gap-2 shrink-0 transition-all ${
              isListening
                ? "bg-[#C0392B] text-white animate-pulse"
                : "bg-[#2D5016] hover:bg-[#4A7C2F] text-white"
            }`}
          >
            {isListening ? <MicOff className="w-4 h-4" /> : <Mic className="w-4 h-4" />}
            {isListening ? "Ouvindo... (Parar)" : "Falar o que mudou"}
          </button>

          <input
            type="text"
            value={voiceTranscript}
            onChange={(e) => setVoiceTranscript(e.target.value)}
            onKeyDown={(e) => e.key === "Enter" && applyVoiceCommand()}
            placeholder='Ex: "Comprei 2 pacotes de café, 6 ovos e acabei com o sabonete"'
            className="flex-1 min-w-0 px-3.5 py-2 rounded-xl bg-white border border-[#DDD8CC] text-xs text-[#1A1F16] focus:outline-hidden focus:border-[#2D5016]"
          />

          <button
            type="button"
            onClick={applyVoiceCommand}
            disabled={loadingVoiceAI || !voiceTranscript.trim()}
            className="px-4 py-2 rounded-xl bg-[#D4821A] hover:bg-[#B86B0E] disabled:opacity-40 text-white text-xs font-bold shrink-0"
          >
            {loadingVoiceAI ? "Atualizando..." : "Atualizar despensa"}
          </button>
        </div>

        {/* FILTROS & BUSCA */}
        <div className="flex flex-col md:flex-row gap-3 pt-1">
          <div className="relative flex-1">
            <Search className="w-4 h-4 text-[#6B7A5E] absolute left-3.5 top-1/2 -translate-y-1/2" />
            <input
              type="text"
              value={busca}
              onChange={(e) => setBusca(e.target.value)}
              placeholder="Filtrar por nome (ex: feijão, rúcula, leite, frango)..."
              className="w-full pl-9 pr-4 py-2 rounded-xl bg-[#FDF8F0] border border-[#DDD8CC] text-xs focus:outline-hidden focus:border-[#2D5016]"
            />
          </div>

          <div className="flex gap-1.5 overflow-x-auto pb-1">
            {["Todos", ...AMBIENTES].map((amb) => (
              <button
                key={amb}
                onClick={() => setAmbienteFiltro(amb)}
                className={`px-3 py-1.5 rounded-xl text-xs font-semibold whitespace-nowrap ${
                  ambienteFiltro === amb
                    ? "bg-[#2D5016] text-white"
                    : "bg-[#F2EDE4] text-[#6B7A5E] hover:text-[#1A1F16]"
                }`}
              >
                {amb}
              </button>
            ))}
          </div>
        </div>

        <div className="flex gap-2 overflow-x-auto pb-1">
          {[
            { id: "todos", label: "Todos" },
            { id: "repor", label: "⚠️ Acabando" },
            { id: "peps", label: "⏳ Vencendo logo" },
            { id: "ruptura", label: "🔄 Costuma faltar" },
            { id: "ok", label: "✅ Em dia" },
          ].map((st) => (
            <button
              key={st.id}
              onClick={() => setStatusFiltro(st.id)}
              className={`px-3 py-1 rounded-full text-xs font-semibold border whitespace-nowrap ${
                statusFiltro === st.id
                  ? "bg-[#EDF5E8] text-[#2D5016] border-[#2D5016]"
                  : "bg-white text-[#6B7A5E] border-[#DDD8CC]"
              }`}
            >
              {st.label}
            </button>
          ))}
        </div>
      </div>

      {/* BANNER INVENTÁRIO PRÉVIO */}
      {inventarioPrevio && (
        <div className="p-4 rounded-2xl bg-[#FEF3DC] border border-[#F2C98A] flex flex-col sm:flex-row sm:items-center justify-between gap-2">
          <div className="text-xs text-[#1A1F16]">
            <strong>📋 Conferindo antes de sair:</strong> marque cada prateleira conforme for conferindo.{" "}
            <span className="font-bold text-[#B86B0E]">
              ({Object.values(checkedPrevio).filter(Boolean).length}/{estoque.length} conferidos)
            </span>
          </div>
          <button
            onClick={() => setCheckedPrevio({})}
            className="text-xs font-bold text-[#B86B0E] underline self-start"
          >
            Limpar Marcações
          </button>
        </div>
      )}

      {/* CARDS DE ESTOQUE */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
        {filteredEstoque.map((item) => {
          const isRepor = Number(item.quantidade) <= Number(item.estoqueMinimo);
          const peps = getPepsBadge(item.validadeDias);
          const isChecked = !!checkedPrevio[item.id];

          return (
            <div
              key={item.id}
              className={`bg-white rounded-2xl p-4 border transition-all flex flex-col justify-between gap-3 ${
                isChecked
                  ? "border-[#4A7C2F] bg-[#EDF5E8]/30"
                  : Number(item.quantidade) === 0
                  ? "border-[#C0392B] bg-[#FDECEA]/20"
                  : isRepor
                  ? "border-[#F2C98A]"
                  : "border-[#DDD8CC]"
              }`}
            >
              <div>
                <div className="flex items-start justify-between gap-2">
                  <div className="flex items-start gap-2 min-w-0">
                    {inventarioPrevio && (
                      <input
                        type="checkbox"
                        checked={isChecked}
                        onChange={() =>
                          setCheckedPrevio((p) => ({ ...p, [item.id]: !p[item.id] }))
                        }
                        className="mt-1 w-4 h-4 accent-[#2D5016] rounded cursor-pointer shrink-0"
                      />
                    )}
                    <div className="min-w-0">
                      <h4 className="font-bold text-sm text-[#1A1F16] leading-snug">
                        {item.nome}
                      </h4>
                      <div className="flex flex-wrap gap-1 mt-1.5">
                        <Badge variant="neutral">{item.ambiente}</Badge>
                        {Number(item.quantidade) === 0 ? (
                          <Badge variant="red">Acabou</Badge>
                        ) : (
                          <Badge variant={peps.variant}>{peps.label}</Badge>
                        )}
                        {isRepor && <Badge variant="amber">⚠️ Repor</Badge>}
                        {item.rupturaCiclo && <Badge variant="red">🔄 Falta fácil</Badge>}
                      </div>
                    </div>
                  </div>

                  <div className="flex items-center gap-1 shrink-0">
                    <button
                      onClick={() => openModal(item)}
                      className="p-1.5 rounded-lg text-[#6B7A5E] hover:bg-[#F2EDE4]"
                      title="Editar"
                    >
                      <Edit3 className="w-3.5 h-3.5" />
                    </button>
                    <button
                      onClick={() =>
                        setEstoque((prev) => prev.filter((i) => i.id !== item.id))
                      }
                      className="p-1.5 rounded-lg text-[#6B7A5E] hover:bg-[#FDECEA] hover:text-[#C0392B]"
                      title="Remover"
                    >
                      <Trash2 className="w-3.5 h-3.5" />
                    </button>
                  </div>
                </div>

                <div className="mt-3 text-xs text-[#6B7A5E] flex items-center justify-between">
                  <span>
                    Mínimo: <strong className="text-[#1A1F16]">{formatQty(item.estoqueMinimo, item.unidade)}</strong>
                  </span>
                  {item.precoMedio > 0 && (
                    <span>Ref: {formatCurrency(item.precoMedio)}</span>
                  )}
                </div>
              </div>

              <div className="pt-2.5 border-t border-[#F2EDE4] flex items-center justify-between gap-2">
                <button
                  onClick={() =>
                    onAddToMarket({
                      nome: item.nome,
                      categoria: item.categoria,
                      quantidade: Math.max(1, Math.ceil(item.estoqueMinimo - item.quantidade) || 1),
                      unidade: item.unidade,
                      precoEstimado: item.precoMedio || 0,
                      motivo: isRepor ? "Abaixo do mínimo" : "Reposição manual",
                      ean: item.ean || "",
                    })
                  }
                  className="text-xs font-bold text-[#2D5016] hover:underline flex items-center gap-1"
                >
                  <ShoppingCart className="w-3.5 h-3.5" /> + Lista
                </button>

                <div className="flex items-center gap-1.5 bg-[#FDF8F0] border border-[#DDD8CC] rounded-xl p-1">
                  <button
                    onClick={() => adjustQty(item.id, -1)}
                    className="w-7 h-7 rounded-lg bg-white border border-[#DDD8CC] flex items-center justify-center hover:bg-[#EAE4D9]"
                  >
                    <Minus className="w-3.5 h-3.5" />
                  </button>
                  <span className="px-2 text-xs font-extrabold text-[#1A1F16] min-w-[54px] text-center">
                    {formatQty(item.quantidade, item.unidade)}
                  </span>
                  <button
                    onClick={() => adjustQty(item.id, 1)}
                    className="w-7 h-7 rounded-lg bg-[#2D5016] text-white flex items-center justify-center hover:bg-[#4A7C2F]"
                  >
                    <Plus className="w-3.5 h-3.5" />
                  </button>
                </div>
              </div>
            </div>
          );
        })}
      </div>

      {/* MODAL CADASTRO / EDIÇÃO MANUAL */}
      <Modal
        open={itemModalOpen}
        onClose={() => setItemModalOpen(false)}
        title={editingItem ? "Editar Item da Despensa" : "Cadastrar Novo Item"}
        subtitle="Configure o local, a quantidade mínima e a validade."
      >
        <form onSubmit={saveFormItem} className="space-y-3.5">
          <div>
            <label className="block text-xs font-semibold mb-1">Nome do Item *</label>
            <input
              required
              type="text"
              value={formItem.nome}
              onChange={(e) => setFormItem({ ...formItem, nome: e.target.value })}
              className="w-full px-3.5 py-2 rounded-xl border border-[#DDD8CC] text-sm"
            />
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
            <div>
              <label className="block text-xs font-semibold mb-1">Ambiente</label>
              <select
                value={formItem.ambiente}
                onChange={(e) => setFormItem({ ...formItem, ambiente: e.target.value })}
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-xs bg-white"
              >
                {AMBIENTES.map((a) => (
                  <option key={a} value={a}>{a}</option>
                ))}
              </select>
            </div>
            <div>
              <label className="block text-xs font-semibold mb-1">Setor Mercado</label>
              <select
                value={formItem.categoria}
                onChange={(e) => setFormItem({ ...formItem, categoria: e.target.value })}
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-xs bg-white"
              >
                {CATEGORIAS_SETOR.map((c) => (
                  <option key={c} value={c}>{c}</option>
                ))}
              </select>
            </div>
          </div>

          <div className="grid grid-cols-3 gap-2 sm:gap-3">
            <div>
              <label className="block text-xs font-semibold mb-1">Quantidade</label>
              <input
                type="number"
                step="0.1"
                value={formItem.quantidade}
                onChange={(e) => setFormItem({ ...formItem, quantidade: e.target.value })}
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-sm"
              />
            </div>
            <div>
              <label className="block text-xs font-semibold mb-1">Unidade</label>
              <select
                value={formItem.unidade}
                onChange={(e) => setFormItem({ ...formItem, unidade: e.target.value })}
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-xs bg-white"
              >
                {UNIDADES.map((u) => (
                  <option key={u} value={u}>{u}</option>
                ))}
              </select>
            </div>
            <div>
              <label className="block text-xs font-semibold mb-1">Estoque Mín.</label>
              <input
                type="number"
                step="0.5"
                value={formItem.estoqueMinimo}
                onChange={(e) => setFormItem({ ...formItem, estoqueMinimo: e.target.value })}
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-sm"
              />
            </div>
          </div>

          <div className="grid grid-cols-2 gap-3">
            <div>
              <label className="block text-xs font-semibold mb-1">Validade (dias)</label>
              <input
                type="number"
                value={formItem.validadeDias}
                onChange={(e) => setFormItem({ ...formItem, validadeDias: e.target.value })}
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-sm"
              />
            </div>
            <div>
              <label className="block text-xs font-semibold mb-1">Preço Médio (R$)</label>
              <input
                type="number"
                step="0.01"
                value={formItem.precoMedio}
                onChange={(e) => setFormItem({ ...formItem, precoMedio: e.target.value })}
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-sm"
              />
            </div>
          </div>

          <label className="flex items-center gap-2 text-xs font-medium cursor-pointer pt-1">
            <input
              type="checkbox"
              checked={formItem.rupturaCiclo}
              onChange={(e) => setFormItem({ ...formItem, rupturaCiclo: e.target.checked })}
              className="w-4 h-4 accent-[#C0392B]"
            />
            Marcar como item que costuma faltar
          </label>

          <div className="flex justify-end gap-2 pt-3 border-t border-[#DDD8CC]">
            <button
              type="button"
              onClick={() => setItemModalOpen(false)}
              className="px-4 py-2 rounded-xl text-xs font-semibold text-[#6B7A5E]"
            >
              Cancelar
            </button>
            <button
              type="submit"
              className="px-5 py-2 rounded-xl bg-[#2D5016] text-white text-xs font-bold"
            >
              Salvar Item
            </button>
          </div>
        </form>
      </Modal>
    </div>
  );
}

// ══════════════════════════════════════════════════════════════════════
// 9. ABA 3 — COZINHA & MARMITAS
//    Freezer, planejador PEPS, receitas, executor guiado (escala + YouTube + baixa de insumos).
// ══════════════════════════════════════════════════════════════════════
function CozinhaMarmitasTab({
  estoque,
  setEstoque,
  marmitas,
  setMarmitas,
  receitas,
  setReceitas,
  focusItem,
  focusRecipeId,
  onFocusConsumed,
  notify,
}) {
  // ── ESTADO LOCAL: receita ativa e modais ────────────────────────────
  const [activeRecipe, setActiveRecipe] = useState(null);
  const [scaleMultiplier, setScaleMultiplier] = useState(1);
  const [completedSteps, setCompletedSteps] = useState({});
  const [modalNovoLote, setModalNovoLote] = useState(false);
  const [modalNovaReceita, setModalNovaReceita] = useState(false);
  const [momentoFiltro, setMomentoFiltro] = useState("Todos");

  // Vindo da tela Início ("Fazer esta receita"): abre direto o passo a passo
  useEffect(() => {
    if (!focusRecipeId) return;
    const rec = receitas.find((r) => r.id === focusRecipeId);
    if (rec) {
      setActiveRecipe(rec);
      setScaleMultiplier(1);
      setCompletedSteps({});
    }
    if (onFocusConsumed) onFocusConsumed();
  }, [focusRecipeId]);

  // ── ESTADO LOCAL: formulário de novo lote congelado ─────────────────
  const [loteForm, setLoteForm] = useState({
    nome: "",
    publico: "Família Completa",
    porcoes: 4,
    validadeDias: 30,
    observacao: "",
    baixaInsumos: true,
    insumosUsados: {},
  });

  // ── ESTADO LOCAL: formulário de nova receita (manual ou voz) ────────
  const [recForm, setRecForm] = useState({
    titulo: "",
    tempo: "30 min",
    rendimentoMarmitas: 4,
    diaCozimento: "Domingo (Batch Cooking)",
    momento: "",
    aproveitamento: "",
    adultos: "",
    crianca10a: "",
    bebe1a6m: "",
    ingredientesTexto: "",
    passosTexto: "",
  });
  const [dictatingRecipe, setDictatingRecipe] = useState(false);

  // ── AÇÕES: freezer (consumir marmita / registrar lote) ──────────────
  const consumeMarmita = (id, delta) => {
    setMarmitas((prev) =>
      prev
        .map((m) =>
          m.id === id
            ? { ...m, porcoes: Math.max(0, Number((Number(m.porcoes) + delta).toFixed(1))) }
            : m
        )
        .filter((m) => m.porcoes > 0)
    );
  };

  // Registrar novo lote no freezer
  const handleSaveLote = (e) => {
    e.preventDefault();
    if (!loteForm.nome.trim()) return;
    setMarmitas((prev) => [
      {
        id: `marm-${Date.now()}`,
        nome: loteForm.nome.trim(),
        publico: loteForm.publico,
        porcoes: Math.max(0.5, Number(loteForm.porcoes) || 1),
        validadeDias: Number(loteForm.validadeDias) || 30,
        observacao: loteForm.observacao.trim(),
      },
      ...prev,
    ]);

    if (loteForm.baixaInsumos) {
      setEstoque((prev) =>
        prev.map((it) => {
          const used = Number(loteForm.insumosUsados[it.id]) || 0;
          if (used <= 0) return it;
          return {
            ...it,
            quantidade: Math.max(0, Number((Number(it.quantidade) - used).toFixed(2))),
          };
        })
      );
    }

    setModalNovoLote(false);
    setLoteForm({
      nome: "",
      publico: "Família Completa",
      porcoes: 4,
      validadeDias: 30,
      observacao: "",
      baixaInsumos: true,
      insumosUsados: {},
    });
    notify("Novo lote congelado adicionado ao Freezer!");
  };

  // ── AÇÕES: concluir receita (baixa insumos + opcionalmente congela marmitas) ───
  const handleFinishRecipe = (addMarmitasToFreezer) => {
    if (!activeRecipe) return;

    setEstoque((prev) =>
      prev.map((item) => {
        const ing = activeRecipe.ingredientes.find(
          (i) =>
            item.nome.toLowerCase().includes(i.nome.toLowerCase()) ||
            i.nome.toLowerCase().includes(item.nome.toLowerCase().split(" ")[0])
        );
        if (!ing) return item;
        const totalDeduct = Number(ing.qtd) * scaleMultiplier;
        return {
          ...item,
          quantidade: Math.max(0, Number((Number(item.quantidade) - totalDeduct).toFixed(2))),
        };
      })
    );

    if (addMarmitasToFreezer) {
      const porcoesGeradas = Math.max(
        1,
        Math.round((activeRecipe.rendimentoMarmitas || 4) * scaleMultiplier)
      );
      setMarmitas((prev) => [
        {
          id: `marm-rec-${Date.now()}`,
          nome: `${activeRecipe.titulo} (${scaleMultiplier}x)`,
          publico: "Família Completa",
          porcoes: porcoesGeradas,
          validadeDias: 30,
          observacao: activeRecipe.adaptacoes?.bebe1a6m || "Preparado pelo passo a passo",
        },
        ...prev,
      ]);
      notify(
        `Receita concluída (${scaleMultiplier}x)! Estoque atualizado e +${porcoesGeradas} marmitas enviadas ao freezer.`
      );
    } else {
      notify(`Receita concluída (${scaleMultiplier}x)! Ingredientes descontados da despensa.`);
    }

    setActiveRecipe(null);
    setCompletedSteps({});
  };

  // ── AÇÕES: ditar e salvar receita ───────────────────────────────────
  const handleDictateRecipe = () => {
    const SpeechRecognition =
      window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SpeechRecognition) {
      notify("Microfone não disponível neste navegador. Preencha os campos manualmente.");
      return;
    }
    const recognition = new SpeechRecognition();
    recognition.lang = "pt-BR";
    recognition.onstart = () => setDictatingRecipe(true);
    recognition.onresult = (event) => {
      const spoken = event.results[0][0].transcript;
      setRecForm((prev) => ({
        ...prev,
        passosTexto: prev.passosTexto ? `${prev.passosTexto}\n${spoken}` : spoken,
      }));
      notify("Trecho transcrito no modo de preparo!");
    };
    recognition.onerror = () => {
      setDictatingRecipe(false);
      notify("Não foi possível captar áudio. Digite o modo de preparo.");
    };
    recognition.onend = () => setDictatingRecipe(false);
    recognition.start();
  };

  const handleSaveNewRecipe = (e) => {
    e.preventDefault();
    if (!recForm.titulo.trim()) return;

    const ingredientesParsed = recForm.ingredientesTexto
      .split("\n")
      .filter((l) => l.trim())
      .map((line) => {
        const match = line.match(/^(\d+(?:[.,]\d+)?)\s*(\w+)?\s+(.*)$/);
        if (match) {
          return {
            qtd: parseFloat(match[1].replace(",", ".")),
            unidade: match[2] || "un",
            nome: match[3].trim(),
          };
        }
        return { qtd: 1, unidade: "un", nome: line.trim() };
      });

    const passosParsed = recForm.passosTexto
      .split("\n")
      .filter((l) => l.trim());

    setReceitas((prev) => [
      {
        id: `rec-${Date.now()}`,
        titulo: recForm.titulo.trim(),
        tempo: recForm.tempo || "30 min",
        rendimentoMarmitas: Number(recForm.rendimentoMarmitas) || 4,
        diaCozimento: recForm.diaCozimento || "Planejado",
        momento: recForm.momento || "",
        queryYoutube: recForm.titulo.trim(),
        aproveitamento: recForm.aproveitamento || "Receita personalizada da família.",
        adaptacoes: {
          adultos: recForm.adultos || "Porção padrão temperada.",
          crianca10a: recForm.crianca10a || "Porção escolar / almoço.",
          bebe1a6m: recForm.bebe1a6m || "Separar antes de adicionar sal.",
        },
        ingredientes:
          ingredientesParsed.length > 0
            ? ingredientesParsed
            : [{ nome: "Ingredientes variados", qtd: 1, unidade: "un" }],
        passos:
          passosParsed.length > 0
            ? passosParsed
            : ["Prepare os ingredientes higienizados e finalize conforme preferência."],
      },
      ...prev,
    ]);

    setModalNovaReceita(false);
    setRecForm({
      titulo: "",
      tempo: "30 min",
      rendimentoMarmitas: 4,
      diaCozimento: "Domingo (Batch Cooking)",
      momento: "",
      aproveitamento: "",
      adultos: "",
      crianca10a: "",
      bebe1a6m: "",
      ingredientesTexto: "",
      passosTexto: "",
    });
    notify("Receita personalizada salva com sucesso!");
  };

  // ── DADOS DERIVADOS: filtro por momento do dia ─────────────────────
  const momentosDisponiveis = [...new Set(receitas.map((r) => r.momento).filter(Boolean))];
  const receitasFiltradas =
    momentoFiltro === "Todos" ? receitas : receitas.filter((r) => r.momento === momentoFiltro);

  // ── RENDER ──────────────────────────────────────────────────────────
  return (
    <div className="space-y-6">
      {/* BLOCO 1: FREEZER & MARMITAS PRONTAS */}
      <div className="bg-white rounded-2xl p-4 sm:p-5 md:p-6 border border-[#DDD8CC] shadow-xs space-y-4">
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
          <div>
            <h2 className="font-serif-editorial text-xl font-bold text-[#2D5016] flex items-center gap-2">
              <Snowflake className="w-5 h-5 text-[#1F5C8B]" />
              Marmitas Prontas no Freezer
              <HelpTip text={AJUDA.freezer} />
            </h2>
            <p className="text-xs text-[#6B7A5E]">
              Refeições prontas que fazem sua semana render mais tempo.
            </p>
          </div>
          <button
            onClick={() => setModalNovoLote(true)}
            className="px-4 py-2 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] text-white text-xs font-semibold flex items-center gap-1.5 self-start"
          >
            <Plus className="w-4 h-4" /> Guardar marmitas
          </button>
        </div>

        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3.5">
          {marmitas.map((m) => (
            <div
              key={m.id}
              className="p-4 rounded-2xl bg-[#FDF8F0] border border-[#DDD8CC] flex flex-col justify-between gap-3"
            >
              <div>
                <div className="flex items-center justify-between gap-2 mb-1">
                  <Badge variant="blue">{m.publico}</Badge>
                  <span className="text-[11px] text-[#6B7A5E]">Val: {m.validadeDias}d</span>
                </div>
                <h4 className="font-bold text-sm text-[#1A1F16]">{m.nome}</h4>
                {m.observacao && (
                  <p className="text-xs text-[#6B7A5E] mt-1">{m.observacao}</p>
                )}
              </div>

              <div className="pt-2.5 border-t border-[#EAE4D9] flex flex-wrap items-center justify-between gap-2">
                <div>
                  <span className="font-serif-editorial text-2xl font-extrabold text-[#2D5016]">
                    {formatQty(m.porcoes)}
                  </span>
                  <span className="text-xs text-[#6B7A5E] ml-1">marmitas</span>
                </div>

                <div className="flex items-center gap-1.5">
                  <button
                    onClick={() => consumeMarmita(m.id, -0.5)}
                    className="px-2 py-1 rounded-lg bg-white border border-[#DDD8CC] text-[11px] font-semibold hover:border-[#D4821A]"
                    title="Consumir meia porção"
                  >
                    -0,5
                  </button>
                  <button
                    onClick={() => consumeMarmita(m.id, -1)}
                    className="px-2.5 py-1 rounded-lg bg-white border border-[#DDD8CC] text-xs font-bold hover:border-[#C0392B] hover:text-[#C0392B]"
                  >
                    -1
                  </button>
                  <button
                    onClick={() => consumeMarmita(m.id, 1)}
                    className="p-1.5 rounded-lg bg-[#2D5016] text-white hover:bg-[#4A7C2F]"
                  >
                    <Plus className="w-3.5 h-3.5" />
                  </button>
                </div>
              </div>
            </div>
          ))}
        </div>
      </div>

      {/* BLOCO 2: PLANEJADOR DE COZIMENTO PEPS & RECEITAS */}
      <div className="bg-white rounded-2xl p-4 sm:p-5 md:p-6 border border-[#DDD8CC] shadow-xs space-y-4">
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
          <div>
            <h2 className="font-serif-editorial text-xl font-bold text-[#2D5016] flex items-center gap-2">
              <Utensils className="w-5 h-5 text-[#D4821A]" />
              O que cozinhar agora
              <HelpTip text={AJUDA.cozinhar} />
            </h2>
            <p className="text-xs text-[#6B7A5E]">
              Receitas para aproveitar o que vence primeiro e render marmitas para a semana.
            </p>
          </div>

          <button
            onClick={() => setModalNovaReceita(true)}
            className="px-4 py-2 rounded-xl bg-[#D4821A] hover:bg-[#B86B0E] text-white text-xs font-semibold flex items-center gap-1.5 self-start"
          >
            <Plus className="w-4 h-4" /> Nova receita
          </button>
        </div>

        {focusItem && (
          <div className="p-3 rounded-xl bg-[#EDF5E8] border border-[#C3DEB0] text-xs text-[#2D5016] font-medium">
            🎯 Vamos usar primeiro: <strong>{focusItem.nome}</strong> ({formatQty(focusItem.quantidade, focusItem.unidade)}). Escolha uma receita abaixo para aproveitar!
          </div>
        )}

        {momentosDisponiveis.length > 0 && (
          <div className="flex gap-1.5 overflow-x-auto pb-1 no-scrollbar">
            {["Todos", ...momentosDisponiveis].map((mo) => (
              <button
                key={mo}
                onClick={() => setMomentoFiltro(mo)}
                className={`px-3 py-1 rounded-full text-xs font-semibold border whitespace-nowrap ${
                  momentoFiltro === mo
                    ? "bg-[#EDF5E8] text-[#2D5016] border-[#2D5016]"
                    : "bg-white text-[#6B7A5E] border-[#DDD8CC]"
                }`}
              >
                {mo === "Todos" ? "Todas" : `${MOMENTOS_RECEITA[mo] || "🍴"} ${mo}`}
              </button>
            ))}
          </div>
        )}

        <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
          {receitasFiltradas.map((rec) => (
            <div
              key={rec.id}
              className="rounded-2xl bg-[#FDF8F0] border border-[#DDD8CC] p-4 flex flex-col justify-between gap-4"
            >
              <div className="space-y-2.5">
                <div className="flex items-center justify-between gap-2 flex-wrap">
                  {rec.momento && <Badge variant="blue">{MOMENTOS_RECEITA[rec.momento] || "🍴"} {rec.momento}</Badge>}
                  <Badge variant="amber">📅 {rec.diaCozimento}</Badge>
                  <Badge variant="green">🍱 Rende {rec.rendimentoMarmitas} marm.</Badge>
                </div>

                <h4 className="font-serif-editorial text-base font-bold text-[#2D5016]">
                  {rec.titulo}
                </h4>

                <p className="text-xs text-[#6B7A5E] leading-relaxed">
                  {rec.aproveitamento}
                </p>

                <div className="space-y-1 text-[11px] pt-1">
                  <div className="p-2 rounded-lg bg-white border border-[#EAE4D9]">
                    <strong>👨‍👩‍👧 Adultos & 10a:</strong> {rec.adaptacoes?.adultos}
                  </div>
                  <div className="p-2 rounded-lg bg-white border border-[#EAE4D9]">
                    <strong>👶 Bebê (1a6m):</strong> {rec.adaptacoes?.bebe1a6m}
                  </div>
                </div>
              </div>

              <button
                onClick={() => {
                  setActiveRecipe(rec);
                  setScaleMultiplier(1);
                  setCompletedSteps({});
                }}
                className="w-full py-2.5 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] text-white text-xs font-bold flex items-center justify-center gap-1.5 shadow-xs"
              >
                <Play className="w-3.5 h-3.5" /> Cozinhar passo a passo
              </button>
            </div>
          ))}
        </div>
      </div>

      {/* MODAL EXECUTOR DE RECEITAS GUIADO (PROPORÇÃO + YOUTUBE + BAIXA) */}
      <Modal
        open={!!activeRecipe}
        onClose={() => setActiveRecipe(null)}
        title={activeRecipe?.titulo || ""}
        subtitle={`Tempo estimado: ${activeRecipe?.tempo} · Rendimento base: ${activeRecipe?.rendimentoMarmitas} marmitas`}
        maxWidth="max-w-2xl"
      >
        {activeRecipe && (
          <div className="space-y-5">
            {/* SELETOR DE PROPORÇÃO & YOUTUBE */}
            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 p-3.5 rounded-xl bg-[#FDF8F0] border border-[#DDD8CC]">
              <div>
                <span className="text-xs font-bold text-[#1A1F16] block mb-1">
                  Rendimento da receita:
                </span>
                <div className="inline-flex flex-wrap max-w-full gap-1 rounded-xl bg-[#EAE4D9] p-1">
                  {[0.5, 1, 2].map((mult) => (
                    <button
                      key={mult}
                      type="button"
                      onClick={() => setScaleMultiplier(mult)}
                      className={`px-2.5 sm:px-3.5 py-1 rounded-lg text-xs font-bold transition-all ${
                        scaleMultiplier === mult
                          ? "bg-[#2D5016] text-white"
                          : "text-[#6B7A5E] hover:text-[#1A1F16]"
                      }`}
                    >
                      {mult === 0.5 ? "0,5x" : `${mult}x`} · {mult === 0.5 ? "Meia receita" : mult === 1 ? "Normal" : "Dobrada"}
                      <span className="block text-[10px] font-medium opacity-80">
                        rende {Math.max(1, Math.round(activeRecipe.rendimentoMarmitas * mult))} marm.
                      </span>
                    </button>
                  ))}
                </div>
              </div>

              <a
                href={`https://www.youtube.com/results?search_query=${encodeURIComponent(
                  activeRecipe.queryYoutube || activeRecipe.titulo
                )}`}
                target="_blank"
                rel="noopener noreferrer"
                className="px-4 py-2.5 rounded-xl bg-[#C0392B] hover:bg-[#A93226] text-white text-xs font-bold flex items-center gap-2 self-start sm:self-center shadow-xs"
              >
                <Youtube className="w-4 h-4" /> Ver vídeo da receita
              </a>
            </div>

            {/* INGREDIENTES RECALCULADOS */}
            <div>
              <h4 className="text-xs font-bold uppercase tracking-wider text-[#2D5016] mb-2">
                Ingredientes para esta receita ({scaleMultiplier}x)
              </h4>
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
                {activeRecipe.ingredientes.map((ing, idx) => {
                  const scaledQtd = Number((ing.qtd * scaleMultiplier).toFixed(2));
                  return (
                    <div
                      key={idx}
                      className="p-2.5 rounded-xl bg-[#EDF5E8]/50 border border-[#C3DEB0] flex items-center justify-between gap-2 text-xs"
                    >
                      <span className="font-medium text-[#1A1F16]">{ing.nome}</span>
                      <span className="font-bold text-[#2D5016] shrink-0 text-right">
                        {formatQty(scaledQtd, ing.unidade)}
                      </span>
                    </div>
                  );
                })}
              </div>
            </div>

            {/* PASSO A PASSO INTERATIVO */}
            <div>
              <h4 className="text-xs font-bold uppercase tracking-wider text-[#2D5016] mb-2">
                Modo de preparo (marque conforme avança)
              </h4>
              <div className="space-y-2">
                {activeRecipe.passos.map((passo, idx) => {
                  const done = !!completedSteps[idx];
                  return (
                    <label
                      key={idx}
                      className={`p-3 rounded-xl border flex items-start gap-3 cursor-pointer transition-all ${
                        done
                          ? "bg-[#EDF5E8] border-[#4A7C2F] text-[#6B7A5E] line-through"
                          : "bg-white border-[#DDD8CC] text-[#1A1F16]"
                      }`}
                    >
                      <input
                        type="checkbox"
                        checked={done}
                        onChange={() =>
                          setCompletedSteps((p) => ({ ...p, [idx]: !p[idx] }))
                        }
                        className="mt-0.5 w-4 h-4 accent-[#2D5016]"
                      />
                      <span className="text-xs leading-relaxed">
                        <strong>Passo {idx + 1}:</strong> {passo}
                      </span>
                    </label>
                  );
                })}
              </div>
            </div>

            {/* AÇÕES DE CONCLUSÃO */}
            <div className="flex flex-col sm:flex-row justify-end gap-2.5 pt-3 border-t border-[#DDD8CC]">
              <button
                type="button"
                onClick={() => handleFinishRecipe(false)}
                className="px-4 py-2.5 rounded-xl bg-[#FDF8F0] border border-[#2D5016] text-[#2D5016] text-xs font-bold hover:bg-[#EDF5E8]"
              >
                Terminei! Descontar ingredientes
              </button>
              <button
                type="button"
                onClick={() => handleFinishRecipe(true)}
                className="px-4 py-2.5 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] text-white text-xs font-bold flex items-center justify-center gap-1.5 text-center"
              >
                <Snowflake className="w-4 h-4" />
                Terminei! Descontar e guardar +
                {Math.max(1, Math.round(activeRecipe.rendimentoMarmitas * scaleMultiplier))} Marmitas
              </button>
            </div>
          </div>
        )}
      </Modal>

      {/* MODAL NOVO LOTE FREEZER */}
      <Modal
        open={modalNovoLote}
        onClose={() => setModalNovoLote(false)}
        title="Guardar marmitas no freezer"
        subtitle="Registre as marmitas prontas e, se quiser, desconte os ingredientes usados."
      >
        <form onSubmit={handleSaveLote} className="space-y-3.5">
          <div>
            <label className="block text-xs font-semibold mb-1">O que você preparou? *</label>
            <input
              required
              type="text"
              value={loteForm.nome}
              onChange={(e) => setLoteForm({ ...loteForm, nome: e.target.value })}
              placeholder="Ex: Feijão Carioca Temperado ou Frango Desfiado"
              className="w-full px-3.5 py-2 rounded-xl border border-[#DDD8CC] text-sm"
            />
          </div>
          <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
            <div>
              <label className="block text-xs font-semibold mb-1">Público</label>
              <select
                value={loteForm.publico}
                onChange={(e) => setLoteForm({ ...loteForm, publico: e.target.value })}
                className="w-full px-2.5 py-2 rounded-xl border border-[#DDD8CC] text-xs bg-white"
              >
                <option value="Família Completa">Família Completa</option>
                <option value="Adultos & 10a">Adultos & 10a</option>
                <option value="Bebê (1a6m)">Bebê (1a6m)</option>
              </select>
            </div>
            <div>
              <label className="block text-xs font-semibold mb-1">Quantas marmitas</label>
              <input
                type="number"
                step="0.5"
                value={loteForm.porcoes}
                onChange={(e) => setLoteForm({ ...loteForm, porcoes: e.target.value })}
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-sm"
              />
            </div>
            <div>
              <label className="block text-xs font-semibold mb-1">Validade (dias)</label>
              <input
                type="number"
                value={loteForm.validadeDias}
                onChange={(e) => setLoteForm({ ...loteForm, validadeDias: e.target.value })}
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-sm"
              />
            </div>
          </div>

          <div className="p-3 rounded-xl bg-[#FDF8F0] border border-[#DDD8CC] space-y-2">
            <label className="flex items-center gap-2 text-xs font-bold text-[#2D5016] cursor-pointer">
              <input
                type="checkbox"
                checked={loteForm.baixaInsumos}
                onChange={(e) =>
                  setLoteForm({ ...loteForm, baixaInsumos: e.target.checked })
                }
                className="w-4 h-4 accent-[#2D5016]"
              />
              Descontar da despensa os ingredientes usados
            </label>
            {loteForm.baixaInsumos && (
              <div className="max-h-36 overflow-y-auto space-y-1.5 pt-2 border-t border-[#EAE4D9]">
                {estoque
                  .filter((e) => Number(e.quantidade) > 0)
                  .slice(0, 12)
                  .map((it) => (
                    <div key={it.id} className="flex items-center justify-between text-xs">
                      <span className="truncate pr-2">
                        {it.nome} ({formatQty(it.quantidade, it.unidade)})
                      </span>
                      <input
                        type="number"
                        step="0.1"
                        placeholder="0"
                        value={loteForm.insumosUsados[it.id] || ""}
                        onChange={(e) =>
                          setLoteForm((p) => ({
                            ...p,
                            insumosUsados: { ...p.insumosUsados, [it.id]: e.target.value },
                          }))
                        }
                        className="w-16 px-2 py-1 rounded-lg border border-[#DDD8CC] bg-white text-right"
                      />
                    </div>
                  ))}
              </div>
            )}
          </div>

          <div className="flex justify-end gap-2 pt-2">
            <button
              type="button"
              onClick={() => setModalNovoLote(false)}
              className="px-4 py-2 rounded-xl text-xs font-semibold text-[#6B7A5E]"
            >
              Cancelar
            </button>
            <button
              type="submit"
              className="px-5 py-2 rounded-xl bg-[#2D5016] text-white text-xs font-bold"
            >
              Guardar no freezer
            </button>
          </div>
        </form>
      </Modal>

      {/* MODAL CRIAR RECEITA (MANUAL OU VOZ) */}
      <Modal
        open={modalNovaReceita}
        onClose={() => setModalNovaReceita(false)}
        title="Nova receita"
        subtitle="Cadastre manualmente ou dite o modo de preparo usando o microfone."
      >
        <form onSubmit={handleSaveNewRecipe} className="space-y-3">
          <div>
            <label className="block text-xs font-semibold mb-1">Nome da Receita *</label>
            <input
              required
              type="text"
              value={recForm.titulo}
              onChange={(e) => setRecForm({ ...recForm, titulo: e.target.value })}
              placeholder="Ex: Bolinho Assado de Cuscuz com Queijo e Tomate"
              className="w-full px-3.5 py-2 rounded-xl border border-[#DDD8CC] text-sm"
            />
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
            <div>
              <label className="block text-xs font-semibold mb-1">Dia de Cozimento</label>
              <input
                type="text"
                value={recForm.diaCozimento}
                onChange={(e) => setRecForm({ ...recForm, diaCozimento: e.target.value })}
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-xs"
              />
            </div>
            <div>
              <label className="block text-xs font-semibold mb-1">Rendimento (Marmitas)</label>
              <input
                type="number"
                value={recForm.rendimentoMarmitas}
                onChange={(e) =>
                  setRecForm({ ...recForm, rendimentoMarmitas: e.target.value })
                }
                className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-xs"
              />
            </div>
          </div>

          <div>
            <label className="block text-xs font-semibold mb-1">Momento (opcional)</label>
            <select
              value={recForm.momento}
              onChange={(e) => setRecForm({ ...recForm, momento: e.target.value })}
              className="w-full px-3 py-2 rounded-xl border border-[#DDD8CC] text-xs bg-white"
            >
              <option value="">Sem momento definido</option>
              {Object.keys(MOMENTOS_RECEITA).map((mo) => (
                <option key={mo} value={mo}>{MOMENTOS_RECEITA[mo]} {mo}</option>
              ))}
            </select>
          </div>

          <div>
            <label className="block text-xs font-semibold mb-1">
              Ingredientes (1 por linha, ex: "2 pct Flocão de Milho Cuscuz")
            </label>
            <textarea
              rows={3}
              value={recForm.ingredientesTexto}
              onChange={(e) =>
                setRecForm({ ...recForm, ingredientesTexto: e.target.value })
              }
              className="w-full p-2.5 rounded-xl border border-[#DDD8CC] text-xs"
            />
          </div>

          <div>
            <div className="flex items-center justify-between mb-1">
              <label className="text-xs font-semibold">Modo de preparo (1 passo por linha)</label>
              <button
                type="button"
                onClick={handleDictateRecipe}
                className={`px-2.5 py-1 rounded-lg text-[11px] font-bold flex items-center gap-1 ${
                  dictatingRecipe
                    ? "bg-[#C0392B] text-white animate-pulse"
                    : "bg-[#EDF5E8] text-[#2D5016]"
                }`}
              >
                <Mic className="w-3 h-3" />
                {dictatingRecipe ? "Ouvindo..." : "Falar o modo de preparo"}
              </button>
            </div>
            <textarea
              rows={3}
              value={recForm.passosTexto}
              onChange={(e) => setRecForm({ ...recForm, passosTexto: e.target.value })}
              className="w-full p-2.5 rounded-xl border border-[#DDD8CC] text-xs"
            />
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-2.5">
            <div>
              <label className="block text-xs font-semibold text-[#2D5016] mb-1">
                Para os adultos e a criança
              </label>
              <input
                type="text"
                value={recForm.adultos}
                onChange={(e) => setRecForm({ ...recForm, adultos: e.target.value })}
                className="w-full px-3 py-1.5 rounded-xl border border-[#DDD8CC] text-xs"
              />
            </div>
            <div>
              <label className="block text-xs font-semibold text-[#1F5C8B] mb-1">
                Para o bebê
              </label>
              <input
                type="text"
                value={recForm.bebe1a6m}
                onChange={(e) => setRecForm({ ...recForm, bebe1a6m: e.target.value })}
                className="w-full px-3 py-1.5 rounded-xl border border-[#DDD8CC] text-xs"
              />
            </div>
          </div>

          <div className="flex justify-end gap-2 pt-2">
            <button
              type="button"
              onClick={() => setModalNovaReceita(false)}
              className="px-4 py-2 rounded-xl text-xs font-semibold text-[#6B7A5E]"
            >
              Cancelar
            </button>
            <button
              type="submit"
              className="px-5 py-2 rounded-xl bg-[#2D5016] text-white text-xs font-bold"
            >
              Salvar Receita
            </button>
          </div>
        </form>
      </Modal>
    </div>
  );
}

// ══════════════════════════════════════════════════════════════════════
// 10. ABA 4 — MERCADO & SCANNER
//    Lista de compras por corredor, carrinho com subtotal, leitor de código de barras (EAN).
// ══════════════════════════════════════════════════════════════════════
function MercadoScannerTab({
  listaCompras,
  setListaCompras,
  estoque,
  setEstoque,
  setCompras,
  setCentroCustos,
  notify,
}) {
  // ── ESTADO LOCAL: formulário de item rápido ─────────────────────────
  const [ordenacao, setOrdenacao] = useState("corredor"); // ver MERCADO_ORDENACOES
  const [recolhidas, setRecolhidas] = useState({}); // { "Corredor": true } = recolhido
  const [novoNome, setNovoNome] = useState("");
  const [novaCat, setNovaCat] = useState("Hortifrúti & Sazonais");
  const [novaQtd, setNovaQtd] = useState(1);
  const [novaUn, setNovaUn] = useState("un");
  const [novoPreco, setNovoPreco] = useState("");
  const [copiedWpp, setCopiedWpp] = useState(false);

  // ── ESTADO LOCAL: scanner de código de barras ───────────────────────
  const [scannerOpen, setScannerOpen] = useState(false);
  const [eanInput, setEanInput] = useState("");
  const [scannedProduct, setScannedProduct] = useState(null);
  const [scanQty, setScanQty] = useState(1);
  const [scanPrice, setScanPrice] = useState("");
  const [cameraActive, setCameraActive] = useState(false);
  const videoRef = useRef(null);
  const streamRef = useRef(null);
  const scanIntervalRef = useRef(null);

  // ── AÇÕES: câmera ───────────────────────────────────────────────────
  // Desliga a câmera com segurança
  const stopCamera = () => {
    if (scanIntervalRef.current) {
      clearInterval(scanIntervalRef.current);
      scanIntervalRef.current = null;
    }
    if (streamRef.current) {
      streamRef.current.getTracks().forEach((t) => t.stop());
      streamRef.current = null;
    }
    setCameraActive(false);
  };

  useEffect(() => {
    return () => stopCamera();
  }, []);

  // CORREÇÃO: o <video> só existe depois de cameraActive=true; antes o stream
  // era ligado a um elemento ainda inexistente e a imagem nunca aparecia.
  useEffect(() => {
    if (cameraActive && videoRef.current && streamRef.current) {
      videoRef.current.srcObject = streamRef.current;
      videoRef.current.play?.().catch(() => {});
    }
  }, [cameraActive]);

  // ── AÇÕES: scanner (buscar EAN, adicionar ao carrinho) ──────────────
  const lookupBarcode = (code) => {
    const clean = String(code).trim();
    if (!clean) return;
    setEanInput(clean);

    const fromEstoque = estoque.find((e) => e.ean === clean);
    const fromLista = listaCompras.find((l) => l.ean === clean);

    if (fromEstoque) {
      setScannedProduct({
        nome: fromEstoque.nome,
        categoria: fromEstoque.categoria,
        unidade: fromEstoque.unidade,
        ean: clean,
      });
      setScanPrice(String(fromEstoque.precoMedio || ""));
      setScanQty(1);
      notify(`Produto encontrado: ${fromEstoque.nome}`);
    } else if (fromLista) {
      setScannedProduct({
        nome: fromLista.nome,
        categoria: fromLista.categoria,
        unidade: fromLista.unidade,
        ean: clean,
      });
      setScanPrice(String(fromLista.precoEstimado || ""));
      setScanQty(fromLista.quantidade || 1);
      notify(`Item da lista encontrado: ${fromLista.nome}`);
    } else {
      setScannedProduct({
        nome: `Produto EAN ${clean}`,
        categoria: "Mercearia & Granel",
        unidade: "un",
        ean: clean,
      });
      setScanPrice("");
      setScanQty(1);
    }
  };

  const startLiveBarcodeCamera = async () => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: { facingMode: "environment" },
      });
      streamRef.current = stream;
      setCameraActive(true); // o efeito acima conecta o stream ao <video>

      if ("BarcodeDetector" in window) {
        const detector = new window.BarcodeDetector({
          formats: ["ean_13", "ean_8", "upc_a", "code_128"],
        });
        scanIntervalRef.current = setInterval(async () => {
          if (videoRef.current && videoRef.current.readyState === 4) {
            try {
              const barcodes = await detector.detect(videoRef.current);
              if (barcodes.length > 0 && barcodes[0].rawValue) {
                lookupBarcode(barcodes[0].rawValue);
                stopCamera();
              }
            } catch {
              // Keep scanning
            }
          }
        }, 600);
      } else {
        notify("Câmera ligada! Se não ler sozinha, digite os números do código abaixo.");
      }
    } catch {
      notify("Não foi possível usar a câmera. Digite os números do código abaixo.");
    }
  };

  const confirmScannedToCart = (e) => {
    e.preventDefault();
    if (!scannedProduct || !scannedProduct.nome.trim()) return;
    const priceNum = Number(scanPrice) || 0;
    const qtyNum = Math.max(0.1, Number(scanQty) || 1);

    setListaCompras((prev) => {
      const idx = prev.findIndex(
        (i) =>
          (scannedProduct.ean && i.ean === scannedProduct.ean) ||
          i.nome.toLowerCase() === scannedProduct.nome.toLowerCase()
      );
      if (idx >= 0) {
        const updated = [...prev];
        updated[idx] = {
          ...updated[idx],
          quantidade: qtyNum,
          precoEstimado: priceNum > 0 ? priceNum : updated[idx].precoEstimado,
          noCarrinho: true,
        };
        return updated;
      }
      return [
        {
          id: `cmp-scan-${Date.now()}`,
          nome: scannedProduct.nome.trim(),
          categoria: scannedProduct.categoria,
          quantidade: qtyNum,
          unidade: scannedProduct.unidade || "un",
          precoEstimado: priceNum,
          noCarrinho: true,
          motivo: "Lido no código de barras",
          ean: scannedProduct.ean || "",
        },
        ...prev,
      ];
    });

    notify(`"${scannedProduct.nome}" adicionado ao carrinho (${formatCurrency(priceNum * qtyNum)})!`);
    setScannedProduct(null);
    setEanInput("");
    setScanPrice("");
  };

  // ── AÇÕES: lista de compras (sincronizar, adicionar, finalizar, WhatsApp) ───
  const syncBelowMinToMarket = () => {
    const criticos = estoque.filter(
      (i) => Number(i.quantidade) <= Number(i.estoqueMinimo) || i.rupturaCiclo
    );
    // CORREÇÃO: calcula os itens novos ANTES de atualizar o estado (mensagem agora correta)
    const nomesNaLista = new Set(listaCompras.map((l) => l.nome.toLowerCase()));
    const novos = criticos.filter((it) => !nomesNaLista.has(it.nome.toLowerCase()));

    if (novos.length > 0) {
      setListaCompras((prev) => [
        ...prev,
        ...novos.map((it) => ({
          id: `cmp-sync-${it.id}-${Date.now()}`,
          nome: it.nome,
          categoria: it.categoria,
          quantidade: Math.max(1, Math.ceil(Number(it.estoqueMinimo) - Number(it.quantidade)) || 1),
          unidade: it.unidade,
          precoEstimado: it.precoMedio || 0,
          noCarrinho: false,
          motivo: it.rupturaCiclo ? "Costuma faltar" : "Acabando",
          ean: it.ean || "",
        })),
      ]);
    }
    notify(
      novos.length > 0
        ? `${novos.length} itens que estavam acabando entraram na lista!`
        : "Tudo o que está acabando já está na sua lista."
    );
  };

  const handleAddManual = (e) => {
    e.preventDefault();
    if (!novoNome.trim()) return;
    setListaCompras((prev) => [
      ...prev,
      {
        id: `cmp-${Date.now()}`,
        nome: novoNome.trim(),
        categoria: novaCat,
        quantidade: Math.max(0.1, Number(novaQtd) || 1),
        unidade: novaUn,
        precoEstimado: Number(novoPreco) || 0,
        noCarrinho: false,
        motivo: "Adicionado manualmente",
      },
    ]);
    setNovoNome("");
    setNovaQtd(1);
    setNovoPreco("");
    notify("Item adicionado à lista!");
  };

  const concluirCompraCarrinho = () => {
    const marcados = listaCompras.filter((i) => i.noCarrinho);
    if (marcados.length === 0) {
      notify("Marque os itens que já estão no carrinho para fechar a compra.");
      return;
    }

    const totalGasto = marcados.reduce(
      (acc, i) => acc + (Number(i.quantidade) || 0) * (Number(i.precoEstimado) || 0),
      0
    );

    // Abastecer Estoque
    setEstoque((prev) => {
      const next = [...prev];
      marcados.forEach((item) => {
        const idx = next.findIndex(
          (e) => e.nome.toLowerCase() === item.nome.toLowerCase()
        );
        if (idx >= 0) {
          next[idx] = {
            ...next[idx],
            quantidade: Number(
              (Number(next[idx].quantidade) + Number(item.quantidade)).toFixed(2)
            ),
            precoMedio: Number(item.precoEstimado) > 0 ? Number(item.precoEstimado) : next[idx].precoMedio,
            validadeDias: next[idx].ambiente === "Geladeira" ? 12 : 120,
          };
        } else {
          next.unshift({
            id: `est-cmp-${Date.now()}-${Math.random().toString(36).slice(2, 5)}`,
            nome: item.nome,
            categoria: item.categoria,
            ambiente: mapCategoriaToAmbiente(item.categoria),
            quantidade: Number(item.quantidade),
            unidade: item.unidade,
            estoqueMinimo: 1,
            validadeDias: 60,
            precoMedio: Number(item.precoEstimado) || 0,
            rupturaCiclo: false,
            sazonalGranel: false,
            ean: item.ean || "",
          });
        }
      });
      return next;
    });

    // Registrar no histórico de compras & Centro de Custos
    const hojeBR = new Date().toLocaleDateString("pt-BR");
    setCompras((prev) => [
      {
        id: `nf-mercado-${Date.now()}`,
        mercado: "Compra pelo Modo Mercado",
        data: hojeBR,
        total: Number(totalGasto.toFixed(2)),
        qtdItens: marcados.length,
        tipo: "Compra via App",
        itens: marcados.map((m) => ({
          nome: m.nome,
          categoria: m.categoria,
          quantidade: Number(m.quantidade),
          unidade: m.unidade,
          precoUnit: Number(m.precoEstimado),
          precoTotal: Number((Number(m.quantidade) * Number(m.precoEstimado)).toFixed(2)),
        })),
      },
      ...prev,
    ]);

    setCentroCustos((prev) => ({
      ...prev,
      gastoAcumuladoMes: Number((prev.gastoAcumuladoMes + totalGasto).toFixed(2)),
    }));

    setListaCompras((prev) => prev.filter((i) => !i.noCarrinho));
    notify(`Compra fechada (${formatCurrency(totalGasto)})! Tudo foi para a despensa e para os gastos do mês.`);
  };

  const copiarWhatsApp = () => {
    const linhas = ["🛒 *LISTA DE MERCADO — RENDE!*", ""];
    CATEGORIAS_SETOR.forEach((cat) => {
      const items = listaCompras.filter((i) => i.categoria === cat);
      if (items.length > 0) {
        linhas.push(`*${cat.toUpperCase()}*`);
        items.forEach((i) => {
          linhas.push(`• ${formatQty(i.quantidade, i.unidade)} — ${i.nome}`);
        });
        linhas.push("");
      }
    });
    copiarTexto(linhas.join("\n")).then((ok) => {
      if (ok) {
        setCopiedWpp(true);
        notify("Lista copiada! É só colar no WhatsApp.");
        setTimeout(() => setCopiedWpp(false), 2500);
      } else {
        notify("Não foi possível copiar automaticamente neste navegador.");
      }
    });
  };

  // ── DADOS DERIVADOS: carrinho e totais ──────────────────────────────
  const itensNoCarrinho = listaCompras.filter((i) => i.noCarrinho);
  const subtotalCarrinho = itensNoCarrinho.reduce(
    (acc, i) => acc + (Number(i.quantidade) || 0) * (Number(i.precoEstimado) || 0),
    0
  );
  const totalEstimadoLista = listaCompras.reduce(
    (acc, i) => acc + (Number(i.quantidade) || 0) * (Number(i.precoEstimado) || 0),
    0
  );

  // Corredores já ordenados conforme "Organizar por" (rápido: a lista tem poucas dezenas de itens)
  const secoes = useMemo(() => {
    const valorItem = (i) => (Number(i.quantidade) || 0) * (Number(i.precoEstimado) || 0);
    const extras = [...new Set(listaCompras.map((i) => i.categoria))].filter(
      (c) => c && !CATEGORIAS_SETOR.includes(c)
    ); // corredores fora da lista padrão também aparecem
    const lista = [...CATEGORIAS_SETOR, ...extras]
      .map((cat) => {
        const items = listaCompras.filter((i) => i.categoria === cat);
        return {
          cat,
          items,
          total: items.reduce((s, i) => s + valorItem(i), 0),
          marcados: items.filter((i) => i.noCarrinho).length,
        };
      })
      .filter((s) => s.items.length > 0);

    const conf = MERCADO_ORDENACOES.find((o) => o.id === ordenacao);
    if (ordenacao === "caro" || ordenacao === "barato") {
      const dir = ordenacao === "caro" ? -1 : 1;
      for (const s of lista) s.items = [...s.items].sort((a, b) => dir * (valorItem(a) - valorItem(b)));
      lista.sort((a, b) => dir * (a.total - b.total));
    } else if (conf && conf.setores) {
      const peso = (cat) => {
        const p = conf.setores.indexOf(cat);
        return p === -1 ? 99 : p;
      };
      lista.sort((a, b) => peso(a.cat) - peso(b.cat)); // estável: o resto segue a ordem padrão
      for (const s of lista) {
        s.items = [...s.items].sort((a, b) => (Number(b.quantidade) || 0) - (Number(a.quantidade) || 0));
      }
    }
    return lista.map((s) => ({ ...s, todosMarcados: s.marcados === s.items.length }));
  }, [listaCompras, ordenacao]);

  // ── AÇÕES: corredores (recolher e marcar todos) ─────────────────────
  const alternarSecao = (cat) => setRecolhidas((p) => ({ ...p, [cat]: !p[cat] }));
  const marcarSecao = (cat, marcar) =>
    setListaCompras((prev) => prev.map((i) => (i.categoria === cat ? { ...i, noCarrinho: marcar } : i)));

  // ── RENDER ──────────────────────────────────────────────────────────
  return (
    <div className="space-y-5">
      {/* CABEÇALHO MODO MERCADO + SCANNER EAN */}
      <div className="bg-white rounded-2xl p-4 sm:p-5 border border-[#DDD8CC] shadow-xs space-y-4">
        <div className="flex flex-col md:flex-row md:items-center justify-between gap-3">
          <div>
            <h2 className="font-serif-editorial text-xl font-bold text-[#2D5016] flex items-center gap-2">
              <ShoppingCart className="w-5 h-5 text-[#2D5016]" />
              Lista & Modo Mercado
              <HelpTip text={AJUDA.mercado} />
            </h2>
            <p className="text-xs text-[#6B7A5E]">
              Organizada por corredor para você passar rápido pelo mercado e acompanhar o valor do carrinho.
            </p>
          </div>

          <div className="flex flex-wrap gap-2">
            <button
              onClick={() => {
                setScannerOpen(true);
                setScannedProduct(null);
              }}
              className="px-4 py-2 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] text-white text-xs font-bold flex items-center gap-1.5 shadow-xs"
            >
              <Barcode className="w-4 h-4" /> Ler código de barras
            </button>

            <button
              onClick={syncBelowMinToMarket}
              className="px-3.5 py-2 rounded-xl bg-[#D4821A] hover:bg-[#B86B0E] text-white text-xs font-semibold flex items-center gap-1.5"
            >
              <RefreshCw className="w-3.5 h-3.5" /> Puxar o que está acabando
            </button>

            <button
              onClick={copiarWhatsApp}
              className="px-3 py-2 rounded-xl bg-[#EDF5E8] border border-[#C3DEB0] text-[#2D5016] text-xs font-semibold flex items-center gap-1.5"
            >
              {copiedWpp ? <Check className="w-3.5 h-3.5" /> : <Copy className="w-3.5 h-3.5" />}
              {copiedWpp ? "Copiado!" : "Enviar no WhatsApp"}
            </button>
          </div>
        </div>

        {/* SUBTOTAL DINÂMICO DO CARRINHO */}
        <div className="p-4 rounded-xl bg-[#FDF8F0] border border-[#DDD8CC] flex flex-col sm:flex-row sm:items-center justify-between gap-3">
          <div>
            <span className="text-xs font-semibold text-[#6B7A5E] block">
              <strong className="text-[#1A1F16]">{itensNoCarrinho.length}</strong> de{" "}
              <strong>{listaCompras.length}</strong> itens marcados · Previsão da lista:{" "}
              <strong>{formatCurrency(totalEstimadoLista)}</strong>
            </span>
            <div className="font-serif-editorial text-xl sm:text-2xl font-extrabold text-[#2D5016] mt-0.5">
              Já no carrinho: {formatCurrency(subtotalCarrinho)}
            </div>
          </div>

          <button
            onClick={concluirCompraCarrinho}
            disabled={itensNoCarrinho.length === 0}
            className="px-5 py-2.5 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] disabled:opacity-40 text-white text-xs font-bold flex items-center gap-2 self-start sm:self-center shadow-xs"
          >
            <CheckCircle2 className="w-4 h-4" />
            Fechar compra e guardar na despensa
          </button>
        </div>

        {/* ADICIONAR ITEM RÁPIDO */}
        <form onSubmit={handleAddManual} className="grid grid-cols-1 sm:grid-cols-12 gap-2">
          <input
            type="text"
            value={novoNome}
            onChange={(e) => setNovoNome(e.target.value)}
            placeholder="Adicionar produto à lista..."
            className="sm:col-span-4 px-3 py-2 rounded-xl border border-[#DDD8CC] text-xs"
          />
          <select
            value={novaCat}
            onChange={(e) => setNovaCat(e.target.value)}
            className="sm:col-span-3 px-2.5 py-2 rounded-xl border border-[#DDD8CC] text-xs bg-white"
          >
            {CATEGORIAS_SETOR.map((c) => (
              <option key={c} value={c}>{c}</option>
            ))}
          </select>
          <div className="sm:col-span-2 flex gap-1">
            <input
              type="number"
              step="0.5"
              value={novaQtd}
              onChange={(e) => setNovaQtd(e.target.value)}
              className="w-1/2 px-2 py-2 rounded-xl border border-[#DDD8CC] text-xs"
            />
            <select
              value={novaUn}
              onChange={(e) => setNovaUn(e.target.value)}
              className="w-1/2 px-1.5 py-2 rounded-xl border border-[#DDD8CC] text-xs bg-white"
            >
              {UNIDADES.map((u) => (
                <option key={u} value={u}>{u}</option>
              ))}
            </select>
          </div>
          <input
            type="number"
            step="0.01"
            value={novoPreco}
            onChange={(e) => setNovoPreco(e.target.value)}
            placeholder="R$ Unit."
            className="sm:col-span-2 px-2.5 py-2 rounded-xl border border-[#DDD8CC] text-xs"
          />
          <button
            type="submit"
            className="sm:col-span-1 py-2 rounded-xl bg-[#2D5016] text-white text-xs font-bold flex items-center justify-center"
          >
            <Plus className="w-4 h-4" />
          </button>
        </form>
      </div>

      {/* ORGANIZAR A LISTA: ordenação + recolher/expandir corredores */}
      <div className="bg-white rounded-2xl border border-[#DDD8CC] shadow-xs p-3 sm:p-4 space-y-2">
        <div className="flex flex-col sm:flex-row sm:items-center gap-2">
          <label htmlFor="ordenacao-mercado" className="text-xs font-semibold text-[#6B7A5E] shrink-0">
            Organizar por
          </label>
          <select
            id="ordenacao-mercado"
            value={ordenacao}
            onChange={(e) => setOrdenacao(e.target.value)}
            className="flex-1 min-w-0 px-3 py-2 rounded-xl border border-[#DDD8CC] text-xs font-semibold bg-white"
          >
            {MERCADO_ORDENACOES.map((o) => (
              <option key={o.id} value={o.id}>{o.label}</option>
            ))}
          </select>
          <div className="flex gap-2 shrink-0">
            <button
              onClick={() => setRecolhidas(Object.fromEntries(secoes.map((s) => [s.cat, true])))}
              className="flex-1 sm:flex-none px-3 py-2 rounded-xl bg-[#FDF8F0] border border-[#DDD8CC] text-[#1A1F16] text-xs font-semibold"
            >
              Recolher tudo
            </button>
            <button
              onClick={() => setRecolhidas({})}
              className="flex-1 sm:flex-none px-3 py-2 rounded-xl bg-[#FDF8F0] border border-[#DDD8CC] text-[#1A1F16] text-xs font-semibold"
            >
              Expandir tudo
            </button>
          </div>
        </div>
        <p className="text-[11px] text-[#6B7A5E]">
          {MERCADO_ORDENACOES.find((o) => o.id === ordenacao)?.dica}
        </p>
      </div>

      {/* CORREDORES DO SUPERMERCADO */}
      <div className="space-y-4">
        {secoes.map(({ cat, items, total, marcados, todosMarcados }) => {
          const recolhida = !!recolhidas[cat];

          return (
            <div
              key={cat}
              className="bg-white rounded-2xl border border-[#DDD8CC] overflow-hidden shadow-xs"
            >
              <div
                className={`px-3 sm:px-4 py-2.5 bg-[#FDF8F0] flex flex-wrap items-center justify-between gap-x-3 gap-y-2 ${
                  recolhida ? "" : "border-b border-[#DDD8CC]"
                }`}
              >
                <button
                  onClick={() => alternarSecao(cat)}
                  aria-expanded={!recolhida}
                  className="flex items-center gap-2 min-w-0 text-left"
                >
                  {recolhida ? (
                    <ChevronDown className="w-4 h-4 text-[#6B7A5E] shrink-0" />
                  ) : (
                    <ChevronUp className="w-4 h-4 text-[#6B7A5E] shrink-0" />
                  )}
                  <h3 className="font-serif-editorial text-sm font-bold text-[#2D5016]">{cat}</h3>
                </button>
                <div className="flex items-center gap-2">
                  <span className="text-[11px] text-[#6B7A5E]">{formatCurrency(total)}</span>
                  <Badge variant={todosMarcados ? "green" : "neutral"}>
                    {marcados}/{items.length}
                  </Badge>
                  <button
                    onClick={() => marcarSecao(cat, !todosMarcados)}
                    className="px-2.5 py-1 rounded-lg bg-[#EDF5E8] border border-[#C3DEB0] text-[#2D5016] text-[11px] font-bold whitespace-nowrap"
                  >
                    {todosMarcados ? "Desmarcar todos" : "Marcar todos"}
                  </button>
                </div>
              </div>

              {!recolhida && (
              <div className="divide-y divide-[#F2EDE4]">
                {items.map((item) => (
                  <div
                    key={item.id}
                    className={`px-4 py-3 flex flex-col sm:flex-row sm:items-center justify-between gap-2.5 ${
                      item.noCarrinho ? "bg-[#EDF5E8]/40" : "bg-white"
                    }`}
                  >
                    <label className="flex items-start gap-3 cursor-pointer flex-1 min-w-0">
                      <input
                        type="checkbox"
                        checked={item.noCarrinho}
                        onChange={() =>
                          setListaCompras((prev) =>
                            prev.map((it) =>
                              it.id === item.id ? { ...it, noCarrinho: !it.noCarrinho } : it
                            )
                          )
                        }
                        className="mt-0.5 w-4 h-4 accent-[#2D5016] rounded cursor-pointer shrink-0"
                      />
                      <div className="min-w-0">
                        <span
                          className={`text-sm font-semibold block ${
                            item.noCarrinho
                              ? "line-through text-[#6B7A5E]"
                              : "text-[#1A1F16]"
                          }`}
                        >
                          {item.nome}
                        </span>
                        {item.motivo && (
                          <span className="text-[11px] text-[#B86B0E]">{suavizarTexto(item.motivo)}</span>
                        )}
                      </div>
                    </label>

                    <div className="flex flex-wrap items-center justify-end gap-2 self-end sm:self-center max-w-full sm:shrink-0">
                      <div className="flex items-center gap-1 bg-[#FDF8F0] border border-[#DDD8CC] rounded-lg px-2 py-1 text-xs">
                        <input
                          type="number"
                          step="0.5"
                          value={item.quantidade}
                          onChange={(e) =>
                            setListaCompras((prev) =>
                              prev.map((it) =>
                                it.id === item.id
                                  ? { ...it, quantidade: Number(e.target.value) }
                                  : it
                              )
                            )
                          }
                          className="w-12 bg-transparent font-bold text-right"
                        />
                        <span className="text-[#6B7A5E]">{item.unidade}</span>
                      </div>

                      <div className="flex items-center gap-1 bg-[#FDF8F0] border border-[#DDD8CC] rounded-lg px-2 py-1 text-xs">
                        <span className="text-[#6B7A5E]">R$</span>
                        <input
                          type="number"
                          step="0.01"
                          value={item.precoEstimado}
                          onChange={(e) =>
                            setListaCompras((prev) =>
                              prev.map((it) =>
                                it.id === item.id
                                  ? { ...it, precoEstimado: Number(e.target.value) }
                                  : it
                              )
                            )
                          }
                          className="w-14 bg-transparent font-bold text-right"
                        />
                      </div>

                      <span className="text-xs font-extrabold text-[#2D5016] min-w-[68px] text-right">
                        {formatCurrency(
                          (Number(item.quantidade) || 0) * (Number(item.precoEstimado) || 0)
                        )}
                      </span>

                      <button
                        onClick={() =>
                          setListaCompras((prev) => prev.filter((it) => it.id !== item.id))
                        }
                        className="p-1 text-[#6B7A5E] hover:text-[#C0392B]"
                      >
                        <X className="w-4 h-4" />
                      </button>
                    </div>
                  </div>
                ))}
              </div>
              )}
            </div>
          );
        })}
      </div>

      {/* MODAL LEITOR DE CÓDIGO DE BARRAS (EAN) */}
      <Modal
        open={scannerOpen}
        onClose={() => {
          stopCamera();
          setScannerOpen(false);
        }}
        title="Ler código de barras"
        subtitle="Aponte a câmera para o código da embalagem ou digite os números."
      >
        <div className="space-y-4">
          {/* CÂMERA AO VIVO */}
          <div className="rounded-2xl bg-[#1A1F16] overflow-hidden relative min-h-[160px] flex flex-col items-center justify-center p-4 text-center">
            {cameraActive ? (
              <>
                <video
                  ref={videoRef}
                  autoPlay
                  playsInline
                  muted
                  className="w-full max-h-52 object-cover rounded-xl"
                />
                <button
                  type="button"
                  onClick={stopCamera}
                  className="mt-2 px-3 py-1 rounded-lg bg-[#C0392B] text-white text-xs font-semibold"
                >
                  Desligar a câmera
                </button>
              </>
            ) : (
              <div className="space-y-2 py-2">
                <Camera className="w-8 h-8 text-[#A8E06B] mx-auto" />
                <p className="text-xs text-white/80">
                  Ligue a câmera traseira para ler o código automaticamente
                </p>
                <button
                  type="button"
                  onClick={startLiveBarcodeCamera}
                  className="px-4 py-2 rounded-xl bg-[#4A7C2F] hover:bg-[#2D5016] text-white text-xs font-bold"
                >
                  Ligar a câmera
                </button>
              </div>
            )}
          </div>

          {/* BUSCA / BIPAGEM MANUAL EAN */}
          <div>
            <label className="block text-xs font-semibold text-[#1A1F16] mb-1">
              Código de barras (ou digite os números):
            </label>
            <div className="flex gap-2">
              <input
                type="text"
                value={eanInput}
                onChange={(e) => setEanInput(e.target.value)}
                onKeyDown={(e) => e.key === "Enter" && lookupBarcode(eanInput)}
                placeholder="Ex: 7898080640611"
                className="flex-1 min-w-0 px-3.5 py-2 rounded-xl border border-[#DDD8CC] text-sm font-mono"
              />
              <button
                type="button"
                onClick={() => lookupBarcode(eanInput)}
                className="px-4 py-2 rounded-xl bg-[#2D5016] text-white text-xs font-bold"
              >
                Buscar produto
              </button>
            </div>

            {/* ATALHOS DE TESTE RÁPIDO COM EANS REAIS DA NOTA */}
            <div className="flex flex-wrap gap-1.5 mt-2">
              <span className="text-[11px] text-[#6B7A5E]">Testar com exemplo:</span>
              {[
                { label: "Leite Italac", ean: "7898080640611" },
                { label: "Sabonete Albany", ean: "7908324408447" },
                { label: "Ovos Líder", ean: "7897207600019" },
              ].map((sample) => (
                <button
                  key={sample.ean}
                  type="button"
                  onClick={() => lookupBarcode(sample.ean)}
                  className="px-2 py-0.5 rounded-md bg-[#F2EDE4] hover:bg-[#EAE4D9] text-[11px] font-medium text-[#1A1F16]"
                >
                  {sample.label}
                </button>
              ))}
            </div>
          </div>

          {/* FORMULÁRIO DO PRODUTO BIPADO */}
          {scannedProduct && (
            <form
              onSubmit={confirmScannedToCart}
              className="p-4 rounded-xl bg-[#EDF5E8] border border-[#C3DEB0] space-y-3"
            >
              <div>
                <label className="block text-[11px] font-bold text-[#2D5016] mb-1">
                  Produto encontrado
                </label>
                <input
                  required
                  type="text"
                  value={scannedProduct.nome}
                  onChange={(e) =>
                    setScannedProduct({ ...scannedProduct, nome: e.target.value })
                  }
                  className="w-full px-3 py-2 rounded-xl border border-[#C3DEB0] bg-white text-sm font-semibold"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-semibold mb-1">Quantidade</label>
                  <input
                    required
                    type="number"
                    step="0.1"
                    value={scanQty}
                    onChange={(e) => setScanQty(e.target.value)}
                    className="w-full px-3 py-2 rounded-xl border border-[#C3DEB0] bg-white text-sm"
                  />
                </div>
                <div>
                  <label className="block text-xs font-semibold mb-1">
                    Preço na prateleira (R$) *
                  </label>
                  <input
                    required
                    autoFocus
                    type="number"
                    step="0.01"
                    value={scanPrice}
                    onChange={(e) => setScanPrice(e.target.value)}
                    placeholder="0,00"
                    className="w-full px-3 py-2 rounded-xl border border-[#C3DEB0] bg-white text-sm font-bold"
                  />
                </div>
              </div>

              <button
                type="submit"
                className="w-full py-2.5 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] text-white text-xs font-bold"
              >
                + Adicionar ao Carrinho ({formatCurrency((Number(scanQty) || 1) * (Number(scanPrice) || 0))})
              </button>
            </form>
          )}
        </div>
      </Modal>
    </div>
  );
}

// ══════════════════════════════════════════════════════════════════════
// 11. ABA 5 — NOTAS FISCAIS
//    Importar nota por foto/PDF/texto (IA), revisar antes de confirmar e histórico de compras.
// ══════════════════════════════════════════════════════════════════════
function NotasFiscaisTab({
  compras,
  setCompras,
  setEstoque,
  setCentroCustos,
  notify,
}) {
  // ── ESTADO LOCAL ────────────────────────────────────────────────────
  const [loadingExtract, setLoadingExtract] = useState(false);
  const [previewUrl, setPreviewUrl] = useState(null);
  const [selectedFile, setSelectedFile] = useState(null);
  const [rawText, setRawText] = useState("");
  const [showTextPanel, setShowTextPanel] = useState(false);
  const [stagingNota, setStagingNota] = useState(null);
  const [expandedNfId, setExpandedNfId] = useState("nf-kacula-2026");
  const fileRef = useRef(null);

  // ── AÇÕES: anexar arquivo, extrair com IA, confirmar importação ─────
  const handleFileUpload = (e) => {
    const file = e.target.files?.[0];
    if (!file) return;
    setSelectedFile(file);
    const reader = new FileReader();
    reader.onload = (ev) => setPreviewUrl(ev.target.result);
    reader.readAsDataURL(file);
    e.target.value = ""; // permite escolher o mesmo arquivo de novo
  };

  const extractFromImage = async () => {
    if (!selectedFile || !previewUrl) return;
    setLoadingExtract(true);
    try {
      const base64Data = previewUrl.split(",")[1];
      const mimeType = selectedFile.type || "image/jpeg";
      const parsed = await extractReceiptWithGemini({ base64Data, mimeType });
      setStagingNota(parsed);
      notify("Nota fiscal extraída com sucesso! Revise os itens abaixo.");
    } catch {
      notify("Erro ao ler a imagem da nota fiscal. Tente uma foto mais nítida ou cole o texto.");
    }
    setLoadingExtract(false);
  };

  const extractFromText = async () => {
    if (!rawText.trim()) return;
    setLoadingExtract(true);
    try {
      const parsed = await extractReceiptWithGemini({ rawText: rawText.trim() });
      setStagingNota(parsed);
      setShowTextPanel(false);
      setRawText("");
      notify("Nota lida! Confira os itens antes de guardar na despensa.");
    } catch {
      notify("Não consegui ler o texto da nota. Confira se copiou a nota inteira.");
    }
    setLoadingExtract(false);
  };

  const confirmImportNota = () => {
    if (!stagingNota) return;
    const totalNota = Number(stagingNota.total) || 0;

    setEstoque((prev) => {
      const next = [...prev];
      (stagingNota.itens || []).forEach((it) => {
        const idx = next.findIndex(
          (e) => e.nome.toLowerCase() === it.nome.toLowerCase()
        );
        if (idx >= 0) {
          next[idx] = {
            ...next[idx],
            quantidade: Number(
              (Number(next[idx].quantidade) + (Number(it.quantidade) || 1)).toFixed(2)
            ),
            precoMedio: Number(it.precoUnit) || next[idx].precoMedio,
          };
        } else {
          next.unshift({
            id: `est-nf-${Date.now()}-${Math.random().toString(36).slice(2, 5)}`,
            nome: it.nome,
            categoria: it.categoria || "Mercearia & Granel",
            ambiente: it.ambiente || mapCategoriaToAmbiente(it.categoria),
            quantidade: Number(it.quantidade) || 1,
            unidade: it.unidade || "un",
            estoqueMinimo: 1,
            validadeDias: it.ambiente === "Geladeira" ? 12 : 90,
            precoMedio: Number(it.precoUnit) || 0,
            rupturaCiclo: false,
            sazonalGranel: false,
          });
        }
      });
      return next;
    });

    setCompras((prev) => [
      {
        id: `nf-${Date.now()}`,
        mercado: stagingNota.mercado || "Supermercado",
        data: stagingNota.data || new Date().toLocaleDateString("pt-BR"),
        total: totalNota,
        qtdItens: stagingNota.itens?.length || 0,
        tipo: "Nota fiscal importada",
        itens: stagingNota.itens || [],
      },
      ...prev,
    ]);

    setCentroCustos((prev) => ({
      ...prev,
      gastoAcumuladoMes: Number((prev.gastoAcumuladoMes + totalNota).toFixed(2)),
    }));

    setStagingNota(null);
    setSelectedFile(null);
    setPreviewUrl(null);
    notify("Nota fiscal salva no histórico e abastecida na despensa!");
  };

  // ── RENDER ──────────────────────────────────────────────────────────
  return (
    <div className="space-y-6">
      {/* IMPORTAÇÃO (FOTO / PDF / TEXTO) */}
      <div className="bg-white rounded-2xl p-4 sm:p-5 md:p-6 border border-[#DDD8CC] shadow-xs space-y-4">
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
          <div>
            <h2 className="font-serif-editorial text-xl font-bold text-[#2D5016] flex items-center gap-2">
              <FileText className="w-5 h-5 text-[#2D5016]" />
              Notas Fiscais & Histórico
              <HelpTip text={AJUDA.notas} />
            </h2>
            <p className="text-xs text-[#6B7A5E]">
              Tire foto do cupom fiscal ou cole o texto da nota para abastecer a despensa sem digitar item por item.
            </p>
          </div>

          <button
            onClick={() => setShowTextPanel(!showTextPanel)}
            className="px-3.5 py-2 rounded-xl bg-[#EDF5E8] border border-[#C3DEB0] text-[#2D5016] text-xs font-semibold self-start"
          >
            {showTextPanel ? "Fechar texto da nota" : "📋 Colar texto da nota"}
          </button>
        </div>

        {showTextPanel && (
          <div className="p-4 rounded-xl bg-[#FDF8F0] border border-[#DDD8CC] space-y-3">
            <textarea
              rows={4}
              value={rawText}
              onChange={(e) => setRawText(e.target.value)}
              placeholder="Cole aqui o texto da nota fiscal..."
              className="w-full p-3 rounded-xl border border-[#DDD8CC] text-xs bg-white"
            />
            <button
              onClick={extractFromText}
              disabled={loadingExtract || !rawText.trim()}
              className="px-4 py-2 rounded-xl bg-[#2D5016] text-white text-xs font-bold flex items-center gap-2"
            >
              <Sparkles className={`w-4 h-4 ${loadingExtract ? "animate-spin" : ""}`} />
              {loadingExtract ? "Extraindo..." : "Processar Texto com Gemini"}
            </button>
          </div>
        )}

        <div
          onClick={() => fileRef.current?.click()}
          className="border-2 border-dashed border-[#DDD8CC] hover:border-[#4A7C2F] rounded-2xl p-6 text-center cursor-pointer bg-[#FDF8F0]"
        >
          {previewUrl ? (
            <div className="space-y-2">
              <p className="text-sm font-bold text-[#2D5016]">
                Arquivo pronto: {selectedFile?.name}
              </p>
              <p className="text-xs text-[#6B7A5E]">Clique para alterar</p>
            </div>
          ) : (
            <div className="space-y-1.5">
              <Camera className="w-8 h-8 text-[#4A7C2F] mx-auto" />
              <p className="text-sm font-semibold text-[#1A1F16]">
                Toque para tirar foto ou anexar o cupom (foto ou PDF)
              </p>
              <p className="text-xs text-[#6B7A5E]">JPG, PNG ou PDF</p>
            </div>
          )}
          <input
            ref={fileRef}
            type="file"
            accept="image/*,application/pdf"
            onChange={handleFileUpload}
            onClick={(e) => e.stopPropagation()}
            className="hidden"
          />
        </div>

        {selectedFile && !stagingNota && (
          <button
            onClick={extractFromImage}
            disabled={loadingExtract}
            className="w-full py-3 rounded-xl bg-[#2D5016] text-white text-xs font-bold flex items-center justify-center gap-2"
          >
            <Sparkles className={`w-4 h-4 ${loadingExtract ? "animate-spin" : ""}`} />
            {loadingExtract ? "Lendo Nota Fiscal com Gemini..." : "Extrair Itens da Nota Fiscal"}
          </button>
        )}
      </div>

      {/* REVISÃO PRÉ-IMPORTAÇÃO */}
      {stagingNota && (
        <div className="bg-white rounded-2xl p-4 sm:p-5 border-2 border-[#4A7C2F] shadow-md space-y-4">
          <div className="flex items-start justify-between gap-3 border-b border-[#DDD8CC] pb-3">
            <div className="min-w-0">
              <Badge variant="green">Confira antes de salvar</Badge>
              <h3 className="font-serif-editorial text-lg font-bold text-[#2D5016] mt-1">
                {stagingNota.mercado}
              </h3>
              <p className="text-xs text-[#6B7A5E]">
                {stagingNota.data} · {stagingNota.itens?.length} itens
              </p>
            </div>
            <span className="font-serif-editorial text-xl sm:text-2xl font-extrabold text-[#D4821A] shrink-0">
              {formatCurrency(stagingNota.total)}
            </span>
          </div>

          <div className="max-h-72 overflow-y-auto divide-y divide-[#F2EDE4]">
            {(stagingNota.itens || []).map((it, idx) => (
              <div key={idx} className="py-2 flex items-center justify-between text-xs gap-2">
                <div className="min-w-0">
                  <strong className="text-[#1A1F16] block">{it.nome}</strong>
                  <span className="text-[#6B7A5E]">
                    {formatQty(it.quantidade, it.unidade)} · {it.ambiente}
                  </span>
                </div>
                <span className="font-bold shrink-0">{formatCurrency(it.precoTotal)}</span>
              </div>
            ))}
          </div>

          <div className="flex justify-end gap-2">
            <button
              onClick={() => setStagingNota(null)}
              className="px-4 py-2 rounded-xl text-xs font-semibold text-[#6B7A5E]"
            >
              Cancelar
            </button>
            <button
              onClick={confirmImportNota}
              className="px-5 py-2 rounded-xl bg-[#2D5016] text-white text-xs font-bold"
            >
              Tudo certo, guardar na despensa
            </button>
          </div>
        </div>
      )}

      {/* HISTÓRICO DE COMPRAS */}
      <div className="space-y-3">
        <h3 className="font-serif-editorial text-lg font-bold text-[#2D5016]">
          Compras anteriores
        </h3>
        {compras.map((nf) => {
          const open = expandedNfId === nf.id;
          return (
            <div
              key={nf.id}
              className="bg-white rounded-2xl border border-[#DDD8CC] overflow-hidden shadow-xs"
            >
              <button
                onClick={() => setExpandedNfId(open ? null : nf.id)}
                className="w-full p-4 flex items-center justify-between gap-3 text-left hover:bg-[#FDF8F0]"
              >
                <div className="min-w-0">
                  <div className="flex items-center gap-2 flex-wrap">
                    <span className="font-bold text-sm text-[#1A1F16]">{nf.mercado}</span>
                    {nf.tipo && (
                      <Badge variant={nf.tipo.includes("Ruptura") ? "amber" : "green"}>
                        {nf.tipo.includes("Ruptura") ? "Compra extra no meio do mês" : nf.tipo}
                      </Badge>
                    )}
                  </div>
                  <p className="text-xs text-[#6B7A5E] mt-0.5">
                    Data: {nf.data} · {nf.qtdItens} itens
                  </p>
                </div>

                <div className="flex items-center gap-2 sm:gap-3 shrink-0">
                  <span className="font-serif-editorial text-base sm:text-lg font-extrabold text-[#2D5016]">
                    {formatCurrency(nf.total)}
                  </span>
                  {open ? (
                    <ChevronUp className="w-4 h-4 text-[#6B7A5E]" />
                  ) : (
                    <ChevronDown className="w-4 h-4 text-[#6B7A5E]" />
                  )}
                </div>
              </button>

              {open && nf.itens && (
                <div className="px-4 pb-4 pt-2 border-t border-[#F2EDE4] bg-[#FDF8F0]/50">
                  <div className="max-h-60 overflow-y-auto divide-y divide-[#EAE4D9]">
                    {nf.itens.map((it, i) => (
                      <div key={i} className="py-2 flex items-center justify-between gap-2 text-xs">
                        <span>
                          {it.nome} ({formatQty(it.quantidade, it.unidade)})
                        </span>
                        <span className="font-bold shrink-0">{formatCurrency(it.precoTotal)}</span>
                      </div>
                    ))}
                  </div>
                </div>
              )}
            </div>
          );
        })}
      </div>
    </div>
  );
}

// ── Campo de valor em R$ com botões − / + (usado na renda e no limite do mercado)
function CampoValor({ rotulo, valor, passo, onChange, children }) {
  const n = Number(valor) || 0;
  return (
    <div className="p-4 rounded-2xl bg-[#FDF8F0] border border-[#DDD8CC] space-y-3">
      <div className="text-xs font-semibold text-[#6B7A5E]">{rotulo}</div>
      <div className="flex items-center justify-between gap-2">
        <button
          type="button"
          onClick={() => onChange(Math.max(0, n - passo))}
          aria-label="Diminuir"
          className="w-10 h-10 rounded-xl bg-white border border-[#DDD8CC] hover:border-[#2D5016] flex items-center justify-center shrink-0 active:scale-95 transition-all"
        >
          <Minus className="w-4 h-4" />
        </button>
        <label className="flex items-baseline justify-center gap-1 min-w-0 flex-1">
          <span className="text-sm font-bold text-[#6B7A5E]">R$</span>
          <input
            type="number"
            inputMode="decimal"
            min="0"
            step={passo}
            value={n}
            onChange={(e) => onChange(Math.max(0, Number(e.target.value) || 0))}
            className="w-full min-w-0 max-w-[9ch] text-center bg-transparent font-serif-editorial text-3xl font-extrabold text-[#2D5016] focus:outline-hidden"
          />
        </label>
        <button
          type="button"
          onClick={() => onChange(n + passo)}
          aria-label="Aumentar"
          className="w-10 h-10 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] text-white flex items-center justify-center shrink-0 active:scale-95 transition-all"
        >
          <Plus className="w-4 h-4" />
        </button>
      </div>
      {children}
    </div>
  );
}

function PerfisDadosTab({
  perfis,
  setPerfis,
  perfilAtivoId,
  onExportDatabaseJson,
  onImportDatabaseJson,
  onResetBaseV2,
  notify,
}) {
  // ── ESTADO LOCAL ────────────────────────────────────────────────────
  const perfil = perfis.find((p) => p.id === perfilAtivoId) || perfis[0];
  const [editandoNome, setEditandoNome] = useState(false);
  const [nomeRascunho, setNomeRascunho] = useState("");
  const [moradorForm, setMoradorForm] = useState(null); // null = janela fechada
  const [novoFoco, setNovoFoco] = useState("");
  const [backupAberto, setBackupAberto] = useState(false);
  const importFileRef = useRef(null);

  // ── DADOS DERIVADOS ─────────────────────────────────────────────────
  const membros = perfil.membros || [];
  const focos = String(perfil.focoNutricional || "")
    .split("·")
    .map((s) => s.trim())
    .filter(Boolean); // o foco do mês é guardado como texto separado por " · "
  const renda = Number(perfil.rendaMensal) || 0;
  const limite = Number(perfil.orcamentoAlimentacao) || 0;
  const pctRenda = renda > 0 ? Math.round((limite / renda) * 100) : null;
  const resumo = Object.keys(TIPOS_MORADOR)
    .map((t) => ({ t, n: membros.filter((m) => classificarMembro(m) === t).length }))
    .filter((x) => x.n > 0);
  const idadeForm =
    moradorForm && moradorForm.idade !== "" && Number.isFinite(Number(moradorForm.idade))
      ? Number(moradorForm.idade)
      : null;
  const tipoPrevio = moradorForm
    ? classificarMembro({ idade: idadeForm, idadeUnidade: moradorForm.idadeUnidade, tipo: "adulto" })
    : null;

  // ── AÇÕES (tudo salva na hora, sem botão "Salvar") ───────────────────
  const atualizar = (patch) =>
    setPerfis((prev) => prev.map((p) => (p.id === perfil.id ? { ...p, ...patch } : p)));

  const salvarNome = () => {
    const n = nomeRascunho.trim();
    if (n) atualizar({ nome: n });
    setEditandoNome(false);
  };

  const abrirMorador = (m) =>
    setMoradorForm(
      m
        ? { id: m.id, nome: m.nome, idade: m.idade ?? "", idadeUnidade: m.idadeUnidade || "anos", emoji: m.emoji || null }
        : { id: null, nome: "", idade: "", idadeUnidade: "anos", emoji: null }
    );

  const salvarMorador = (e) => {
    e.preventDefault();
    if (!moradorForm.nome.trim()) return;
    const novo = {
      id: moradorForm.id || `mem-${Date.now()}`,
      nome: moradorForm.nome.trim(),
      idade: idadeForm === null ? null : Math.max(0, idadeForm),
      idadeUnidade: moradorForm.idadeUnidade,
      tipo: tipoPrevio,
      emoji: moradorForm.emoji,
    };
    atualizar({
      membros: moradorForm.id ? membros.map((m) => (m.id === novo.id ? novo : m)) : [...membros, novo],
    });
    setMoradorForm(null);
  };

  const removerMorador = (id) => {
    atualizar({ membros: membros.filter((m) => m.id !== id) });
    setMoradorForm(null);
    notify("Morador removido.");
  };

  const salvarFocos = (lista) => atualizar({ focoNutricional: lista.join(" · ") });
  const temFoco = (t) => focos.some((f) => f.toLowerCase() === t.toLowerCase());
  const alternarFoco = (t) =>
    salvarFocos(temFoco(t) ? focos.filter((f) => f.toLowerCase() !== t.toLowerCase()) : [...focos, t]);
  const adicionarFoco = (e) => {
    e.preventDefault();
    const t = novoFoco.trim().replace(/·/g, "-");
    if (t && !temFoco(t)) salvarFocos([...focos, t]);
    setNovoFoco("");
  };

  // ── RENDER ──────────────────────────────────────────────────────────
  return (
    <div className="space-y-5">
      {/* NOSSA CASA: nome editável + moradores como "figurinhas" */}
      <div
        className="rounded-2xl p-4 sm:p-6 text-white shadow-lg"
        style={{ background: `linear-gradient(135deg, ${COLORS.green} 0%, ${COLORS.greenMid} 100%)` }}
      >
        <div className="flex items-start justify-between gap-3">
          <div className="min-w-0 flex-1">
            <p className="text-[11px] uppercase tracking-wider text-white/70 font-semibold">Nossa casa</p>
            {editandoNome ? (
              <input
                autoFocus
                value={nomeRascunho}
                onChange={(e) => setNomeRascunho(e.target.value)}
                onBlur={salvarNome}
                onKeyDown={(e) => {
                  if (e.key === "Enter") salvarNome();
                  if (e.key === "Escape") setEditandoNome(false);
                }}
                className="mt-1 w-full px-3 py-2 rounded-xl bg-white/15 border border-white/30 text-white font-serif-editorial text-2xl font-extrabold focus:outline-hidden"
              />
            ) : (
              <button
                onClick={() => {
                  setNomeRascunho(perfil.nome);
                  setEditandoNome(true);
                }}
                className="mt-1 flex items-center gap-2 text-left max-w-full group"
                title="Toque para mudar o nome"
              >
                <h2 className="font-serif-editorial text-2xl md:text-3xl font-extrabold tracking-tight truncate">
                  {perfil.nome}
                </h2>
                <Edit3 className="w-4 h-4 text-white/60 group-hover:text-white shrink-0" />
              </button>
            )}
          </div>
          <span className="shrink-0 text-[11px] text-white/80 bg-white/10 px-2.5 py-1 rounded-full">
            ✓ Salva sozinho
          </span>
        </div>

        <div className="mt-4 flex flex-wrap gap-3">
          {membros.map((m) => {
            const tipo = classificarMembro(m);
            return (
              <button
                key={m.id}
                onClick={() => abrirMorador(m)}
                className="flex flex-col items-center gap-1 w-[72px] active:scale-95 transition-all"
                title="Toque para editar"
              >
                <span className="w-14 h-14 rounded-full bg-white/15 hover:bg-white/25 border border-white/25 flex items-center justify-center text-3xl">
                  {emojiDoMembro(m)}
                </span>
                <span className="text-[11px] font-semibold truncate max-w-full">{m.nome}</span>
                <span className="text-[10px] text-white/70 -mt-1">{TIPOS_MORADOR[tipo].rotulo}</span>
              </button>
            );
          })}
          <button
            onClick={() => abrirMorador(null)}
            className="flex flex-col items-center gap-1 w-[72px] active:scale-95 transition-all"
          >
            <span className="w-14 h-14 rounded-full border-2 border-dashed border-white/50 hover:bg-white/10 flex items-center justify-center">
              <Plus className="w-5 h-5" />
            </span>
            <span className="text-[11px] font-semibold">Adicionar</span>
          </button>
        </div>

        {resumo.length > 0 && (
          <p className="mt-3 text-xs text-white/80">
            {resumo.map((x) => `${x.n} ${TIPOS_MORADOR[x.t].plural[x.n === 1 ? 0 : 1]}`).join(" · ")}
          </p>
        )}
      </div>

      {/* BOLSO: renda e limite mensal com botões − / + e barra */}
      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
        <CampoValor
          rotulo="💰 Limite mensal para mercado"
          valor={limite}
          passo={50}
          onChange={(v) => atualizar({ orcamentoAlimentacao: v })}
        >
          <input
            type="range"
            min="0"
            max={Math.max(5000, Math.ceil(limite / 500) * 500)}
            step="50"
            value={limite}
            onChange={(e) => atualizar({ orcamentoAlimentacao: Number(e.target.value) })}
            className="w-full accent-[#2D5016]"
            aria-label="Limite mensal para mercado"
          />
          <p className="text-[11px] text-[#6B7A5E]">
            {pctRenda !== null ? `Equivale a ${pctRenda}% da renda mensal.` : "Informe a renda mensal para ver o percentual."}
          </p>
        </CampoValor>

        <CampoValor
          rotulo="🏦 Renda mensal"
          valor={renda}
          passo={100}
          onChange={(v) => atualizar({ rendaMensal: v })}
        >
          <div className="h-2 bg-[#EAE4D9] rounded-full overflow-hidden">
            <div
              className="h-full rounded-full transition-all"
              style={{ width: `${Math.min(100, pctRenda || 0)}%`, backgroundColor: COLORS.amber }}
            />
          </div>
          <p className="text-[11px] text-[#6B7A5E]">A barra mostra quanto da renda vai para o mercado.</p>
        </CampoValor>
      </div>

      {/* FOCO DO MÊS: etiquetas que se toca para ligar/desligar */}
      <div className="bg-white rounded-2xl p-4 sm:p-5 border border-[#DDD8CC] shadow-xs space-y-3">
        <div>
          <h3 className="font-serif-editorial text-lg font-bold text-[#2D5016] flex items-center gap-2">
            🎯 Foco do mês
          </h3>
          <p className="text-xs text-[#6B7A5E]">O que a casa quer cuidar agora. Aparece na tela Início.</p>
        </div>

        <div className="flex flex-wrap gap-2">
          {focos.length === 0 && <span className="text-xs text-[#6B7A5E]">Nenhum foco ainda. Escolha abaixo ou escreva o seu.</span>}
          {focos.map((f) => (
            <span
              key={f}
              className="pl-3 pr-1.5 py-1.5 rounded-full bg-[#EDF5E8] border border-[#C3DEB0] text-xs font-semibold text-[#2D5016] flex items-center gap-1"
            >
              {f}
              <button
                onClick={() => alternarFoco(f)}
                aria-label={`Remover ${f}`}
                className="w-5 h-5 rounded-full hover:bg-[#C3DEB0] flex items-center justify-center"
              >
                <X className="w-3 h-3" />
              </button>
            </span>
          ))}
        </div>

        <div className="flex flex-wrap gap-1.5">
          {FOCO_SUGESTOES.filter((s) => !temFoco(s)).map((s) => (
            <button
              key={s}
              onClick={() => alternarFoco(s)}
              className="px-2.5 py-1 rounded-full border border-dashed border-[#DDD8CC] text-[11px] text-[#6B7A5E] hover:border-[#2D5016] hover:text-[#2D5016]"
            >
              + {s}
            </button>
          ))}
        </div>

        <form onSubmit={adicionarFoco} className="flex gap-2">
          <input
            value={novoFoco}
            onChange={(e) => setNovoFoco(e.target.value)}
            placeholder="Escrever outro foco…"
            className="flex-1 min-w-0 px-3 py-2 rounded-xl border border-[#DDD8CC] text-xs focus:outline-hidden focus:border-[#2D5016]"
          />
          <button
            type="submit"
            disabled={!novoFoco.trim()}
            className="px-4 py-2 rounded-xl bg-[#2D5016] disabled:opacity-40 text-white text-xs font-bold"
          >
            Adicionar
          </button>
        </form>
      </div>

      {/* CÓPIA DE SEGURANÇA (recolhida para não parecer formulário) */}
      <div className="bg-white rounded-2xl border border-[#DDD8CC] shadow-xs overflow-hidden">
        <button
          onClick={() => setBackupAberto(!backupAberto)}
          aria-expanded={backupAberto}
          className="w-full px-4 sm:px-5 py-4 flex items-center justify-between gap-3 text-left"
        >
          <span className="font-serif-editorial text-lg font-bold text-[#2D5016] flex items-center gap-2">
            <Layers className="w-5 h-5 text-[#D4821A]" />
            Cópia de segurança dos dados
          </span>
          {backupAberto ? <ChevronUp className="w-4 h-4 text-[#6B7A5E]" /> : <ChevronDown className="w-4 h-4 text-[#6B7A5E]" />}
        </button>
        {backupAberto && (
          <div className="px-4 sm:px-5 pb-5 space-y-3">
            <p className="text-xs text-[#6B7A5E]">
              Baixe uma cópia de tudo o que está no Rende! para guardar no celular, no computador ou no Google Drive, e restaure quando quiser continuar de outro aparelho.
            </p>
            <div className="flex flex-wrap gap-3">
              <button
                onClick={onExportDatabaseJson}
                className="px-4 py-2.5 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] text-white text-xs font-bold flex items-center gap-2"
              >
                <Download className="w-4 h-4" /> Exportar cópia
              </button>
              <button
                onClick={() => importFileRef.current?.click()}
                className="px-4 py-2.5 rounded-xl bg-[#FDF8F0] border border-[#DDD8CC] hover:border-[#2D5016] text-[#1A1F16] text-xs font-bold flex items-center gap-2"
              >
                <Upload className="w-4 h-4" /> Restaurar cópia
              </button>
              <input
                ref={importFileRef}
                type="file"
                accept=".json,application/json"
                onChange={onImportDatabaseJson}
                className="hidden"
              />
              <button
                onClick={onResetBaseV2}
                className="px-4 py-2.5 rounded-xl bg-[#FDECEA] border border-[#F5B7B1] text-[#C0392B] text-xs font-bold flex items-center gap-2"
              >
                <RefreshCw className="w-4 h-4" /> Voltar aos dados de exemplo
              </button>
            </div>
          </div>
        )}
      </div>

      {/* JANELA: ADICIONAR / EDITAR MORADOR */}
      <Modal
        open={!!moradorForm}
        onClose={() => setMoradorForm(null)}
        title={moradorForm && moradorForm.id ? "Editar morador" : "Novo morador"}
        subtitle="O Rende! descobre sozinho se é adulto, criança, bebê ou idoso pela idade."
        maxWidth="max-w-md"
      >
        {moradorForm && (
          <form onSubmit={salvarMorador} className="space-y-4">
            <div className="flex items-center gap-3">
              <span className="w-16 h-16 rounded-full bg-[#EDF5E8] border border-[#C3DEB0] flex items-center justify-center text-4xl shrink-0">
                {moradorForm.emoji || TIPOS_MORADOR[tipoPrevio].emoji}
              </span>
              <div className="min-w-0">
                <div className="text-sm font-bold text-[#1A1F16]">
                  {idadeForm === null ? "Informe a idade" : TIPOS_MORADOR[tipoPrevio].rotulo}
                </div>
                <div className="text-[11px] text-[#6B7A5E]">
                  {idadeForm === null ? "Sem idade, será tratado como adulto." : "Definido automaticamente pela idade."}
                </div>
              </div>
            </div>

            <div>
              <label className="block text-xs font-semibold mb-1">Nome *</label>
              <input
                autoFocus
                required
                value={moradorForm.nome}
                onChange={(e) => setMoradorForm({ ...moradorForm, nome: e.target.value })}
                placeholder="Ex: Lucas"
                className="w-full px-3.5 py-2.5 rounded-xl border border-[#DDD8CC] text-sm focus:outline-hidden focus:border-[#2D5016]"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold mb-1">Idade</label>
              <div className="flex gap-2">
                <input
                  type="number"
                  inputMode="numeric"
                  min="0"
                  value={moradorForm.idade}
                  onChange={(e) => setMoradorForm({ ...moradorForm, idade: e.target.value })}
                  className="w-24 px-3 py-2.5 rounded-xl border border-[#DDD8CC] text-sm focus:outline-hidden focus:border-[#2D5016]"
                />
                <div className="inline-flex rounded-xl bg-[#EAE4D9] p-1">
                  {["anos", "meses"].map((u) => (
                    <button
                      key={u}
                      type="button"
                      onClick={() => setMoradorForm({ ...moradorForm, idadeUnidade: u })}
                      className={`px-3.5 py-1 rounded-lg text-xs font-bold transition-all ${
                        moradorForm.idadeUnidade === u ? "bg-[#2D5016] text-white" : "text-[#6B7A5E]"
                      }`}
                    >
                      {u}
                    </button>
                  ))}
                </div>
              </div>
            </div>

            <div>
              <label className="block text-xs font-semibold mb-1">Emoji</label>
              <div className="flex flex-wrap gap-1.5">
                <button
                  type="button"
                  onClick={() => setMoradorForm({ ...moradorForm, emoji: null })}
                  className={`px-3 h-10 rounded-xl text-xs font-bold border ${
                    !moradorForm.emoji ? "bg-[#EDF5E8] border-[#2D5016] text-[#2D5016]" : "bg-white border-[#DDD8CC] text-[#6B7A5E]"
                  }`}
                >
                  Automático
                </button>
                {EMOJIS_MORADORES.map((em) => (
                  <button
                    key={em}
                    type="button"
                    onClick={() => setMoradorForm({ ...moradorForm, emoji: em })}
                    className={`w-10 h-10 rounded-xl text-xl border flex items-center justify-center active:scale-95 ${
                      moradorForm.emoji === em ? "bg-[#EDF5E8] border-[#2D5016]" : "bg-white border-[#DDD8CC]"
                    }`}
                  >
                    {em}
                  </button>
                ))}
              </div>
            </div>

            <div className="flex items-center justify-between gap-2 pt-1">
              {moradorForm.id ? (
                <button
                  type="button"
                  onClick={() => removerMorador(moradorForm.id)}
                  className="px-3 py-2 rounded-xl text-xs font-semibold text-[#C0392B] hover:bg-[#FDECEA] flex items-center gap-1.5"
                >
                  <Trash2 className="w-3.5 h-3.5" /> Remover
                </button>
              ) : (
                <span />
              )}
              <div className="flex gap-2">
                <button
                  type="button"
                  onClick={() => setMoradorForm(null)}
                  className="px-4 py-2 rounded-xl text-xs font-semibold text-[#6B7A5E]"
                >
                  Cancelar
                </button>
                <button
                  type="submit"
                  className="px-5 py-2 rounded-xl bg-[#2D5016] hover:bg-[#4A7C2F] text-white text-xs font-bold"
                >
                  Salvar
                </button>
              </div>
            </div>
          </form>
        )}
      </Modal>
    </div>
  );
}

// ══════════════════════════════════════════════════════════════════════
// 13. APP PRINCIPAL
//    Estado global (db), persistência no localStorage, ações entre abas e layout (cabeçalho + menu + abas).
// ══════════════════════════════════════════════════════════════════════
export default function App() {
  // ── ESTADO GLOBAL ───────────────────────────────────────────────────
  const [activeTab, setActiveTab] = useState("inicio");
  const [despensaFilter, setDespensaFilter] = useState("todos");
  const [cookFocusItem, setCookFocusItem] = useState(null);
  const [cookFocusRecipeId, setCookFocusRecipeId] = useState(null); // receita a abrir ao entrar na Cozinha
  const [toast, setToast] = useState(null);
  const [loadingAudit, setLoadingAudit] = useState(false);

  // ── BANCO DE DADOS: carrega do localStorage e salva a cada alteração ───
  const [db, setDb] = useState(() => {
    try {
      const saved = localStorage.getItem(STORAGE_KEY);
      if (saved) {
        const parsed = JSON.parse(saved);
        if (parsed && Array.isArray(parsed.estoque) && Array.isArray(parsed.perfis)) {
          return normalizeDatabase(parsed); // completa campos ausentes de versões antigas
        }
      }
    } catch {
      // Fallback to initial v2 database
    }
    return createInitialDatabase();
  });

  useEffect(() => {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(db));
    } catch {
      // Ignore quota exceeded
    }
  }, [db]);

  // ── NOTIFICAÇÕES (toast) ────────────────────────────────────────────
  const notify = (msg) => {
    setToast(msg);
    setTimeout(() => {
      setToast((curr) => (curr === msg ? null : curr));
    }, 3600);
  };

  const perfilAtivo =
    db.perfis.find((p) => p.id === db.perfilAtivoId) || db.perfis[0];

  // ── ATUALIZADORES DE ESTADO (um por coleção do banco) ───────────────
  const setEstoque = (up) =>
    setDb((p) => ({ ...p, estoque: typeof up === "function" ? up(p.estoque) : up }));
  const setMarmitas = (up) =>
    setDb((p) => ({ ...p, marmitas: typeof up === "function" ? up(p.marmitas) : up }));
  const setReceitas = (up) =>
    setDb((p) => ({ ...p, receitas: typeof up === "function" ? up(p.receitas) : up }));
  const setListaCompras = (up) =>
    setDb((p) => ({
      ...p,
      listaCompras: typeof up === "function" ? up(p.listaCompras) : up,
    }));
  const setCompras = (up) =>
    setDb((p) => ({ ...p, compras: typeof up === "function" ? up(p.compras) : up }));
  const setCentroCustos = (up) =>
    setDb((p) => ({
      ...p,
      centro_custos: typeof up === "function" ? up(p.centro_custos) : up,
    }));
  const setPerfis = (up) =>
    setDb((p) => ({ ...p, perfis: typeof up === "function" ? up(p.perfis) : up }));
  const setGasBotijao = (up) =>
    setDb((p) => ({ ...p, gasBotijao: typeof up === "function" ? up(p.gasBotijao) : up }));

  // ── AÇÕES ENTRE ABAS ────────────────────────────────────────────────
  const handleConsumeQuick = (item) => {
    const step = item.unidade === "kg" || item.unidade === "L" || item.unidade === "pct" ? 0.25 : 1;
    setEstoque((prev) =>
      prev.map((it) =>
        it.id === item.id
          ? {
              ...it,
              quantidade: Math.max(0, Number((Number(it.quantidade) - step).toFixed(2))),
            }
          : it
      )
    );
    notify(`Baixa rápida registrada em "${item.nome}".`);
  };

  const handleNavigateToCook = (item, recipeId = null) => {
    setCookFocusItem(item);
    setCookFocusRecipeId(recipeId);
    setActiveTab("cozinha");
  };

  const handleAddToMarket = (newItem) => {
    setListaCompras((prev) => {
      const idx = prev.findIndex(
        (i) => i.nome.toLowerCase() === newItem.nome.toLowerCase()
      );
      if (idx >= 0) {
        const copy = [...prev];
        copy[idx] = {
          ...copy[idx],
          quantidade: Number(
            (Number(copy[idx].quantidade) + Number(newItem.quantidade)).toFixed(2)
          ),
        };
        return copy;
      }
      return [
        ...prev,
        {
          id: `cmp-${Date.now()}-${Math.random().toString(36).slice(2, 5)}`,
          nome: newItem.nome,
          categoria: newItem.categoria || "Mercearia & Granel",
          quantidade: Number(newItem.quantidade) || 1,
          unidade: newItem.unidade || "un",
          precoEstimado: Number(newItem.precoEstimado) || 0,
          noCarrinho: false,
          motivo: newItem.motivo || "Reposição",
          ean: newItem.ean || "",
        },
      ];
    });
    notify(`"${newItem.nome}" enviado para Mercado & Scanner!`);
  };

  const handleRegisterMissingItem = ({ nome, categoria, quantidade, unidade }) => {
    // Zera no estoque o(s) item(ns) cujo nome contém o texto digitado.
    // CORREÇÃO: exige pelo menos 3 letras (antes, digitar "a" zerava quase tudo).
    const termo = nome.trim().toLowerCase();
    if (termo.length >= 3) {
      setEstoque((prev) =>
        prev.map((it) =>
          it.nome.toLowerCase().includes(termo) ? { ...it, quantidade: 0 } : it
        )
      );
    }
    handleAddToMarket({
      nome,
      categoria,
      quantidade,
      unidade,
      precoEstimado: 0,
      motivo: "Item faltante registrado no Início",
    });
  };

  const handleAddAllRupturaToMarket = () => {
    const rupturas = db.estoque.filter((i) => i.rupturaCiclo);
    rupturas.forEach((r) =>
      handleAddToMarket({
        nome: r.nome,
        categoria: r.categoria,
        quantidade: Math.max(1, Number(r.estoqueMinimo) - Number(r.quantidade)),
        unidade: r.unidade,
        precoEstimado: r.precoMedio,
        motivo: "Reforço: costuma faltar",
        ean: r.ean,
      })
    );
    setActiveTab("mercado");
  };

  // ── AÇÕES: auditoria IA, exportar/importar/restaurar banco ──────────
  const handleRunAudit = async () => {
    setLoadingAudit(true);
    try {
      const result = await runGeminiFamilyAudit(
        perfilAtivo,
        db.estoque,
        db.marmitas,
        db.centro_custos
      );
      setDb((prev) => ({ ...prev, analiseIA: result }));
      notify("Dicas prontas! Veja o que sugerimos para a casa.");
    } catch {
      notify("Não foi possível conectar à IA Gemini neste momento.");
    }
    setLoadingAudit(false);
  };

  const handleExportDatabaseJson = () => {
    const payload = {
      ...db,
      exportadoEm: new Date().toISOString(),
    };
    const blob = new Blob([JSON.stringify(payload, null, 2)], {
      type: "application/json",
    });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = "database.json";
    a.click();
    URL.revokeObjectURL(url);
    notify("Arquivo database.json exportado com sucesso!");
  };

  const handleImportDatabaseJson = (e) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (ev) => {
      try {
        const parsed = JSON.parse(ev.target.result);
        if (!parsed || !Array.isArray(parsed.estoque)) {
          throw new Error("Estrutura inválida");
        }
        setDb(normalizeDatabase(parsed));
        notify("Base database.json restaurada com sucesso!");
      } catch {
        notify("Arquivo JSON inválido.");
      }
    };
    reader.readAsText(file);
    e.target.value = "";
  };

  const handleResetBaseV2 = () => {
    setDb(createInitialDatabase());
    notify("Dados de exemplo restaurados!");
  };

  // ── DADOS DERIVADOS: contador do menu (itens a repor) ───────────────
  const countRepor = db.estoque.filter(
    (i) => Number(i.quantidade) <= Number(i.estoqueMinimo)
  ).length;

  // ── RENDER (layout: toast → cabeçalho → menu de abas → conteúdo) ────
  return (
    <div className="min-h-screen bg-[#F7F3EC] text-[#1A1F16] font-sans-ui pb-16">
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Fraunces:ital,opsz,wght@0,9..144,400;0,9..144,600;0,9..144,800;1,9..144,400&family=DM+Sans:wght@400;500;600;700&display=swap');
        .font-serif-editorial { font-family: 'Fraunces', Georgia, serif; }
        .font-sans-ui { font-family: 'DM Sans', -apple-system, BlinkMacSystemFont, sans-serif; }
        /* Esconde a barra de rolagem do menu de abas no celular (continua rolando com o dedo) */
        .no-scrollbar::-webkit-scrollbar { display: none; }
        .no-scrollbar { -ms-overflow-style: none; scrollbar-width: none; }
        /* Evita zoom por toque duplo em botões e ajuste automático de fonte no iOS */
        html { -webkit-text-size-adjust: 100%; }
        button, a { touch-action: manipulation; }
      `}</style>

      {/* TOAST NOTIFICATION */}
      {toast && (
        <div className="fixed bottom-4 left-4 right-4 sm:left-auto sm:right-5 sm:bottom-5 z-50 sm:max-w-sm bg-[#1A1F16] text-white px-4 py-3 rounded-2xl shadow-xl border border-white/15 text-xs font-medium flex items-center gap-2.5">
          <CheckCircle2 className="w-4 h-4 text-[#7BAF5A] shrink-0" />
          <span>{toast}</span>
        </div>
      )}

      {/* HEADER EDITORIAL */}
      <header className="sticky top-0 z-30 bg-[#2D5016] text-white shadow-md">
        <div className="max-w-6xl mx-auto px-3 sm:px-4 py-3 sm:py-3.5 flex flex-wrap items-center justify-between gap-2 sm:gap-3">
          <div className="flex items-baseline gap-2.5">
            <h1 className="font-serif-editorial text-xl sm:text-2xl font-extrabold tracking-tight text-white">
              Rende!
            </h1>
            <span className="px-2 py-0.5 rounded-full bg-white/15 text-[10px] font-bold tracking-wider uppercase">
              v2.0
            </span>
          </div>

          <div className="flex items-center gap-2">
            <span className="hidden md:inline text-xs text-white/80 mr-2">
              {perfilAtivo.nome}
            </span>
            <button
              onClick={handleExportDatabaseJson}
              className="px-3 py-1.5 rounded-xl bg-white/10 hover:bg-white/20 text-xs font-semibold flex items-center gap-1.5"
              title="Baixar cópia de segurança dos dados"
              aria-label="Baixar cópia de segurança dos dados"
            >
              <Download className="w-3.5 h-3.5" />
              <span className="hidden sm:inline">Cópia de segurança</span>
            </button>
          </div>
        </div>

        {/* NAVEGAÇÃO 6 ABAS */}
        <nav className="bg-[#234010] border-t border-white/10">
          <div className="max-w-6xl mx-auto px-1 sm:px-4 flex overflow-x-auto no-scrollbar">
            {TABS.map((t) => {
              const Icon = t.icon;
              const isActive = activeTab === t.id;
              return (
                <button
                  key={t.id}
                  onClick={() => {
                    setActiveTab(t.id);
                    if (t.id !== "despensa") setDespensaFilter("todos");
                  }}
                  className={`flex-none sm:flex-1 py-3 px-3.5 sm:px-3 flex items-center justify-center gap-1.5 text-xs font-semibold border-b-2 transition-all whitespace-nowrap ${
                    isActive
                      ? "border-[#F2A84E] text-white bg-white/10"
                      : "border-transparent text-white/70 hover:text-white hover:bg-white/5"
                  }`}
                >
                  <Icon className="w-4 h-4 shrink-0" />
                  {/* label curto no celular/tablet, completo só em telas largas */}
                  <span className="lg:hidden">{t.short}</span>
                  <span className="hidden lg:inline">{t.label}</span>
                  {t.id === "despensa" && countRepor > 0 && (
                    <span className="px-1.5 py-px rounded-full bg-[#D4821A] text-white text-[10px] font-bold">
                      {countRepor}
                    </span>
                  )}
                  {t.id === "mercado" && db.listaCompras.length > 0 && (
                    <span className="px-1.5 py-px rounded-full bg-[#7BAF5A] text-[#1A1F16] text-[10px] font-bold">
                      {db.listaCompras.length}
                    </span>
                  )}
                </button>
              );
            })}
          </div>
        </nav>
      </header>

      {/* CONTEÚDO PRINCIPAL */}
      <main className="max-w-6xl mx-auto px-3 sm:px-4 pt-4 sm:pt-6">
        {activeTab === "inicio" && (
          <InicioTab
            perfilAtivo={perfilAtivo}
            estoque={db.estoque}
            marmitas={db.marmitas}
            receitas={db.receitas}
            centroCustos={db.centro_custos}
            gasBotijao={db.gasBotijao}
            setGasBotijao={setGasBotijao}
            notify={notify}
            analiseIA={db.analiseIA}
            onRunAudit={handleRunAudit}
            loadingAudit={loadingAudit}
            onConsumeQuick={handleConsumeQuick}
            onNavigateToCook={handleNavigateToCook}
            onRegisterMissingItem={handleRegisterMissingItem}
            onAddAllRupturaToMarket={handleAddAllRupturaToMarket}
            onGoToTab={(tab, filter = "todos") => {
              setDespensaFilter(filter);
              setActiveTab(tab);
            }}
          />
        )}

        {activeTab === "despensa" && (
          <DespensaVozTab
            estoque={db.estoque}
            setEstoque={setEstoque}
            initialFilter={despensaFilter}
            onAddToMarket={handleAddToMarket}
            notify={notify}
          />
        )}

        {activeTab === "cozinha" && (
          <CozinhaMarmitasTab
            estoque={db.estoque}
            setEstoque={setEstoque}
            marmitas={db.marmitas}
            setMarmitas={setMarmitas}
            receitas={db.receitas}
            setReceitas={setReceitas}
            focusItem={cookFocusItem}
            focusRecipeId={cookFocusRecipeId}
            onFocusConsumed={() => setCookFocusRecipeId(null)}
            notify={notify}
          />
        )}

        {activeTab === "mercado" && (
          <MercadoScannerTab
            listaCompras={db.listaCompras}
            setListaCompras={setListaCompras}
            estoque={db.estoque}
            setEstoque={setEstoque}
            setCompras={setCompras}
            setCentroCustos={setCentroCustos}
            notify={notify}
          />
        )}

        {activeTab === "notas" && (
          <NotasFiscaisTab
            compras={db.compras}
            setCompras={setCompras}
            setEstoque={setEstoque}
            setCentroCustos={setCentroCustos}
            notify={notify}
          />
        )}

        {activeTab === "perfis" && (
          <PerfisDadosTab
            perfis={db.perfis}
            setPerfis={setPerfis}
            perfilAtivoId={db.perfilAtivoId}
            onExportDatabaseJson={handleExportDatabaseJson}
            onImportDatabaseJson={handleImportDatabaseJson}
            onResetBaseV2={handleResetBaseV2}
            notify={notify}
          />
        )}
      </main>
    </div>
  );
}
